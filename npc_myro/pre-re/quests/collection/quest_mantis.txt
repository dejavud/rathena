lou_fild01,240,354,4	script	螳螂研究员	878,{

	if (checkquest(60179) != -1) {
		if (checkquest(60179,HUNTING) != 2 ) {
			mes "[螳螂研究员]";
			mes "你猎完 50 只螳螂了吗？";
			next;
			switch(select("是:不:我想放弃")) {
			case 1:
				mes "[螳螂研究员]";
				mes "嗯，我看你还没猎够 50 只。";
				close;
			case 2:
				mes "[螳螂研究员]";
				mes "记住，我需要帮忙猎杀螳螂，去猎 50 只吧。";
				close;
			case 3:
				mes "[螳螂研究员]";
				mes "你确定要停止狩猎吗？";
				mes "你目前的进度都会被清除";
				next;
				switch(select("是:不")) {
				case 1:
					mes "[螳螂研究员]";
					mes "那好吧，如果你改变主意就回这里来。";
					erasequest 60179;
					close;
				case 2:
					mes "[螳螂研究员]";
					mes "请去杀 50 只螳螂，让这里重新成为人们可以安心练级的地方。";
					close;
				}
			}
		}
		mes "[螳螂研究员]";
		mes "厉害，你完成得真快。";
		getexp 18818,7527;
		erasequest 60179;
		close;
	}
	if ((BaseLevel > 34) && (BaseLevel < 71)) {
		mes "[螳螂研究员]";
		mes "哦太好了！";
		mes "你是来帮我的对吧？";
		mes "如果不是，我真不知道拿这里的怪物怎么办才好！";
		next;
		switch(select("当然:不")) {
		case 1:
			mes "[螳螂研究员]";
			mes "太好了！我一见到你就知道可以指望你！";
			mes "我需要你减少这张地图上正在肆虐的怪物数量！";
			mes "这附近螳螂太多，低等级的冒险者根本没法在这里好好练级。";
			next;
			mes "[螳螂研究员]";
			mes "那么，你还愿意帮我吗？";
			next;
			switch(select("当然:取消")) {
			case 1:
				if (BaseLevel > 70) {
					mes "[螳螂研究员]";
					mes "多谢你帮我解决了螳螂的大麻烦！";
					mes "目前情况稳定了，不再需要你帮忙猎杀了。";
					close;
				}
				mes "[螳螂研究员]";
				mes "去杀 50 只螳螂，完成后告诉我，我会酬谢你的付出。";
				setquest 60179;
				close;
			case 2:
				mes "[螳螂研究员]";
				mes "如果你改变主意，请再回来。";
				close;
			}
		case 2:
			mes "[螳螂研究员]";
			mes "如果你改变主意，请再回来。";
			close;
		}
	}
	mes "[螳螂研究员]";
	mes "哦天哪，哦天哪！";
	next;
	mes "[螳螂研究员]";
	mes "抱歉我现在没法聊天，我快惨死了，这些该死的螳螂快把我杀了。";
	close;
}
