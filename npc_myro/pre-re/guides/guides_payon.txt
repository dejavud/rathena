payon,162,67,4	script	解说员#pay	708,{
	cutin "pay_soldier",2;
	mes "[斐扬解说员]";
	mes "欢迎来到山岳之都斐扬。";
	mes "如果您对这一带不熟悉，";
	mes "我可以帮您找到想去的地方。";
	set .@loop1,1;
	while (.@loop1) {
		next;
		switch(select("城市导览:删除小地图上的标记:注意事项。:取消")) {
		case 1:
			mes "[斐扬解说员]";
			mes "请从菜单中选择您想去的地点。";
			if (.@compass_check == 0) {
				mes "需要我把这些地点";
				mes "标记到您的小地图上吗？";
				next;
				if (select("是。:否。") == 1) set .@compass_check,1;
			}
			set .@loop2,1;
			while(.@loop2) {
				if (.@wait_button_chk == 0) set .@wait_button_chk,1;
				else next;
				switch(select("^FF0000弓箭手公会^000000:武器店:道具店:酒馆:中央宫:皇后宫:宫殿分馆:御膳房:锻造坊:取消")) {
				case 1:
					mes "[斐扬解说员]";
					mes "弓箭手公会负责弓箭手职业的转职，";
					mes "位于斐扬东北方的弓箭手村里。";
					if (.@compass_check)
						viewpoint 1,227,328,2,0xFF0000;
					break;
				case 2:
					mes "[斐扬解说员]";
					mes "武器店在斐扬的西北角。";
					if (.@compass_check)
						viewpoint 1,139,159,3,0xFF00FF;
					break;
				case 3:
					mes "[斐扬解说员]";
					mes "道具店在斐扬靠近西北角的地方。";
					if (.@compass_check)
						viewpoint 1,144,85,4,0xFF00FF;
					break;
				case 4:
					mes "[斐扬解说员]";
					mes "酒馆在斐扬的东北方向，";
					mes "是漫长狩猎之后放松的好去处。";
					if (.@compass_check)
						viewpoint 1,220,117,5,0xFF00FF;
					break;
				case 5:
					mes "[斐扬解说员]";
					mes "中央宫位于斐扬的北部。";
					if (.@compass_check)
						viewpoint 1,155,245,6,0x00FF00;
					break;
				case 6:
					mes "[斐扬解说员]";
					mes "皇后宫在斐扬的西北方向。";
					if (.@compass_check)
						viewpoint 1,107,324,7,0x00FF00;
					break;
				case 7:
					mes "[斐扬解说员]";
					mes "宫殿分馆在斐扬的西部。";
					if (.@compass_check)
						viewpoint 1,130,204,8,0x00FF00;
					break;
				case 8:
					mes "[斐扬解说员]";
					mes "御膳房位于斐扬靠近北端的地方。";
					if (.@compass_check)
						viewpoint 1,154,325,9,0x00FF00;
					break;
				case 9:
					mes "[斐扬解说员]";
					mes "锻造坊就在斐扬的中心附近。";
					if (.@compass_check)
						viewpoint 1,126,169,10,0xFFFF00;
					break;
				case 10:
					mes "[斐扬解说员]";
					mes "如果不想让小地图继续显示位置标记，";
					mes "请选择''删除小地图上的标记''。";
					set .@loop2,0;
					break;
				}
			}
			break;
		case 2:
			viewpoint 2,237,41,2,0x00FF00;
			viewpoint 2,237,41,3,0x0000FF;
			viewpoint 2,46,345,4,0xFF00FF;
			viewpoint 2,175,220,5,0xFF0000;
			viewpoint 2,175,220,6,0xFF0000;
			viewpoint 2,175,220,7,0xFF0000;
			viewpoint 2,237,41,8,0x0000FF;
			viewpoint 2,46,345,9,0x00FF00;
			viewpoint 2,175,220,10,0xFF0000;
			set .@compass_check,0;
			break;
		case 3:
			mes "[斐扬解说员]";
			mes "借助魔法与技术的进步，";
			mes "我们的导览系统得到升级，";
			mes "可以把地点标记到您的小地图上，";
			mes "方便您在城市里认路。";
			next;
			mes "[斐扬解说员]";
			mes "小地图在屏幕右上角。";
			mes "如果看不到，请按 Ctrl + Tab，";
			mes "或点击基本信息窗口里的''地图''按钮。";
			next;
			mes "[斐扬解说员]";
			mes "点击小地图上的''+''和''-''";
			mes "可以放大或缩小显示范围。";
			mes "祝您在斐扬旅途愉快。";
			break;
		case 4:
			mes "[斐扬解说员]";
			mes "勇敢的冒险者，";
			mes "祝您旅途平安。";
			close2;
			set .@loop1,0;
			break;
		}
	}
	cutin "",255;
	end;
}
