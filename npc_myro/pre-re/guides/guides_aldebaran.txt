aldebaran,139,63,4	script	导览兵#alde	105,{
	cutin "prt_soldier",2;
	mes "[艾尔帕兰导览兵]";
	mes "我只是个普通士兵，";
	mes "随便在哪座城市都能见到的那种，";
	mes "恐怕连个名字都没有……";
	next;
	mes "[艾尔帕兰导览兵]";
	mes "我负责艾尔帕兰守备队的导览服务，";
	mes "就由我来带你逛逛这座城市吧！";
	next;
	if (select("查看地点导览。:结束对话。") == 1) {
		viewpoint 1,61,229,0,0xFF6633;
		viewpoint 1,72,197,1,0x0000FF;
		viewpoint 1,223,222,2,0x00FFFF;
		viewpoint 1,233,105,3,0x515151;
		viewpoint 1,197,70,4,0x3355FF;
		viewpoint 1,60,60,5,0xFF5555;
		mes "^FF6633+^000000 -> 卡普拉总部 ";
		mes "^0000FF+^000000 -> 武器店 ";
		mes "^00FFFF+^000000 -> 贤者公会（关闭）";
		mes "^515151+^000000 -> 酒馆 ";
		mes "^3355FF+^000000 -> 道具店 ";
		mes "^FF5555+^000000 -> 炼金术士公会";
		close2;
		cutin "prt_soldier",255;
		end;
	}
	mes "[艾尔帕兰导览兵]";
	mes "我们誓死守卫艾尔帕兰！";
	mes "愿正义的铁拳永远粉碎邪恶！";
	close2;
	cutin "prt_soldier",255;
	end;
}
