alberta,23,238,4	script	解说员#alb::AlbGuide	105,{
	cutin "prt_soldier",2;
	mes "[艾尔贝塔解说员]";
	mes "欢迎来到港口之都艾尔贝塔。";
	mes "如果您在城里找不到地方，";
	mes "或需要人带路，请随时告诉我。";
	set .@loop1,1;
	while (.@loop1) {
		next;
		switch(select("城市导览:删除小地图上的标记:注意事项:取消")) {
		case 1:
			mes "[艾尔贝塔解说员]";
			mes "请从菜单中选择您想去的地点。";
			if (.@compass_check == 0) {
				mes "需要我把这些地点";
				mes "标记到您的小地图上吗？";
				next;
				if (select("是:否") == 1) set .@compass_check,1;
			}
			set .@loop2,1;
			while (.@loop2) {
				if (.@wait_button_chk == 0) set .@wait_button_chk,1;
				else next;

				switch(select("^FF0000商人公会^000000:武器店:道具店:旅馆:锻造坊:取消")) {
				case 1:
					mes "[艾尔贝塔解说员]";
					mes "商人公会负责商人职业的转职，";
					mes "位于艾尔贝塔的西南角。";
					if (.@compass_check)
						viewpoint 1,33,41,2,0xFF0000;
					break;
				case 2:
					mes "[艾尔贝塔解说员]";
					mes "武器店在艾尔贝塔的南端。";
					if (.@compass_check)
						viewpoint 1,117,37,3,0xFF00FF;
					break;
				case 3:
					mes "[艾尔贝塔解说员]";
					mes "道具店离艾尔贝塔的中心不远，";
					mes "应该很好找。";
					if (.@compass_check)
						viewpoint 1,98,154,4,0xFF00FF;
					break;
				case 4:
					mes "[艾尔贝塔解说员]";
					mes "艾尔贝塔的北端有一家旅馆，";
					mes "您可以在那里休息。";
					if (.@compass_check)
						viewpoint 1,65,233,5,0xFF00FF;
					break;
				case 5:
					mes "[艾尔贝塔解说员]";
					mes "锻造坊和商人公会在同一栋建筑里，";
					mes "就在西南边。";
					if (.@compass_check)
						viewpoint 1,35,41,6,0xFF00FF;
					break;
				case 6:
					mes "[艾尔贝塔解说员]";
					mes "如果不想让小地图继续显示位置标记，";
					mes "请选择''删除小地图上的标记''。";
					set .@loop2,0;
				}
			}
			break;
		case 2:
			viewpoint 2,237,41,2,0xFF0000;
			viewpoint 2,237,41,3,0xFF00FF;
			viewpoint 2,46,345,4,0xFF00FF;
			viewpoint 2,175,220,5,0xFF00FF;
			viewpoint 2,175,220,6,0xFF00FF;
			set .@compass_check,0;
			break;
		case 3:
			mes "[艾尔贝塔解说员]";
			mes "借助魔法与技术的进步，";
			mes "我们的导览系统得到升级，";
			mes "可以把地点标记到您的小地图上，";
			mes "方便您在城市里认路。";
			next;
			mes "[艾尔贝塔解说员]";
			mes "小地图在屏幕右上角。";
			mes "如果看不到，请按 Ctrl + Tab，";
			mes "或点击基本信息窗口里的''地图''按钮。";
			next;
			mes "[艾尔贝塔解说员]";
			mes "点击小地图上的''+''和''-''";
			mes "可以放大或缩小显示范围。";
			mes "祝您在艾尔贝塔旅途愉快。";
			break;
		case 4:
			mes "[艾尔贝塔解说员]";
			mes "祝您旅途平安。";
			mes "关于艾尔贝塔还有什么问题，";
			mes "欢迎随时来问我。";
			close2;
			set .@loop1,0;
		}
	}
	cutin "prt_soldier",255;
	end;
}

alberta,120,60,3	duplicate(AlbGuide)	解说员#2alb	105
