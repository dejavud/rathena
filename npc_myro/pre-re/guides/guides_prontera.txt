prontera,154,187,4	script	解说员#prt::PrtGuide	105,{
	cutin "prt_soldier",2;
	mes "[普隆德拉解说员]";
	mes "欢迎来到卢恩·米德加兹王国";
	mes "美丽的首都普隆德拉。";
	mes "如果您有什么疑问，";
	mes "或在城里找不到地方，请随时问我。";
	set .@loop1,1;
	while (.@loop1) {
		next;
		switch(select("城市导览:删除小地图上的标记:注意事项:取消")) {
		case 1:
			mes "[普隆德拉解说员]";
			mes "请从下面的菜单中选择您想知道的地点。";
			if (.@compass_check == 0) {
				mes "需要我在小地图上标记位置吗？";
				next;
				if (select("是:否") == 1) set .@compass_check,1;
			}
			set .@loop2,1;
			while (.@loop2) {
				if (.@wait_button_chk == 0) set .@wait_button_chk,1;
				else next;

				switch(select("剑士公会:^0000FF教堂^000000:普隆德拉骑士团:武器店:道具店:旅馆:交易所:酒馆:图书馆:职业介绍所:普隆德拉城:市政厅:取消")) {
				case 1:
					mes "[普隆德拉解说员]";
					mes "剑士公会负责剑士的转职业务，";
					mes "不过已经搬到依斯鲁得岛去了，";
					mes "现在那里只是一栋空建筑。";
					if (.@compass_check)
						viewpoint 1,237,41,4,0x00FF00;
					break;
				case 2:
					mes "[普隆德拉解说员]";
					mes "普隆德拉教堂负责服事的转职业务，";
					mes "位于城市的东北角。";
					if (.@compass_check)
						viewpoint 1,236,316,5,0xFF0000;
					break;
				case 3:
					mes "[普隆德拉解说员]";
					mes "普隆德拉骑士团负责首都的治安，";
					mes "位于城市的西北角。";
					if (.@compass_check)
						viewpoint 1,46,345,6,0x00FF00;
					break;
				case 4:
					mes "[普隆德拉解说员]";
					mes "武器店位于中央喷水池的东北方向。";
					if (.@compass_check)
						viewpoint 1,175,220,7,0xFF00FF;
					break;
				case 5:
					mes "[普隆德拉解说员]";
					mes "道具店位于中央喷水池的西北方向。";
					if (.@compass_check)
						viewpoint 1,134,221,8,0xFF00FF;
					break;
				case 6:
					mes "[普隆德拉解说员]";
					mes "普隆德拉的旅馆有两家，";
					mes "分别在中央喷水池一带的东边和西边。";
					if (.@compass_check) {
						viewpoint 1,204,191,9,0xFF00FF;
						viewpoint 1,107,192,10,0xFF00FF;
					}
					break;
				case 7:
					mes "[普隆德拉解说员]";
					mes "交易所位于中央喷水池的东南方向。";
					if (.@compass_check)
						viewpoint 1,179,184,11,0x00FF00;
					break;
				case 8:
					mes "[普隆德拉解说员]";
					mes "酒馆在喷水池的东南方，";
					mes "就在交易所的后面。";
					if (.@compass_check)
						viewpoint 1,208,154,12,0x00FF00;
					break;
				case 9:
					mes "[普隆德拉解说员]";
					mes "从中央喷水池一直往北走有片空地，";
					mes "普隆德拉图书馆的两个分馆";
					mes "分别在空地的东边和西边。";
					if (.@compass_check) {
						viewpoint 1,120,267,13,0x00FF00;
						viewpoint 1,192,267,14,0x00FF00;
					}
					break;
				case 10:
					mes "[普隆德拉解说员]";
					mes "职业介绍所就在中央喷水池的西南方。";
					if (.@compass_check)
						viewpoint 1,133,183,15,0x00FF00;
					break;
				case 11:
					mes "[普隆德拉解说员]";
					mes "普隆德拉城位于本市的北部，";
					mes "从城堡的后门出去，";
					mes "可以前往城市北边的原野。";
					if (.@compass_check)
						viewpoint 1,156,360,16,0x00FF00;
					break;
				case 12:
					mes "[普隆德拉解说员]";
					mes "市政厅位于普隆德拉的西南角。";
					if (.@compass_check)
						viewpoint 1,75,91,17,0x01FF01;
					break;
				case 13:
					mes "[普隆德拉解说员]";
					mes "如果不想让小地图继续显示位置标记，";
					mes "请选择''删除小地图上的标记''。";
					set .@loop2,0;
					break;
				}
			}
			break;
		case 2:
			viewpoint 2,237,41,4,0x00FF00;
			viewpoint 2,237,41,5,0x0000FF;
			viewpoint 2,46,345,6,0x00FF00;
			viewpoint 2,175,220,7,0xFF0000;
			viewpoint 2,134,221,8,0xFF0000;
			viewpoint 2,204,191,9,0xFF0000;
			viewpoint 2,107,192,10,0xFF0000;
			viewpoint 2,179,184,11,0x00FF00;
			viewpoint 2,208,154,12,0x00FF00;
			viewpoint 2,120,267,13,0x00FF00;
			viewpoint 2,192,267,14,0x00FF00;
			viewpoint 2,133,183,15,0x00FF00;
			viewpoint 2,156,360,16,0x00FF00;
			viewpoint 2,75,91,17,0x00FF00;
			set .@compass_check,0;
			break;
		case 3:
			mes "[普隆德拉解说员]";
			mes "魔法与技术的进步，";
			mes "让我们的信息系统得以升级，";
			mes "可以在小地图上标记地点，出行更方便。";
			next;
			mes "[普隆德拉解说员]";
			mes "小地图在屏幕右上角，";
			mes "如果看不到，请按 Ctrl + Tab 键，";
			mes "或点击基本信息窗口里的''地图''按钮。";
			next;
			mes "[普隆德拉解说员]";
			mes "点击小地图上的''+''和''-''，";
			mes "可以放大或缩小地图。";
			mes "祝您在普隆德拉旅途愉快。";
			break;
		case 4:
			mes "[普隆德拉解说员]";
			mes "那么，冒险者……";
			mes "祝您在卢恩·米德加兹王国的旅程";
			mes "既开心又平安。";
			close2;
			set .@loop1,0;
			break;
		}
	}
	cutin "prt_soldier",255;
	end;
}
