			mes "按礼数还是先去拜见城主大人为好。";

			break;
		case 2:
			mes "[士兵]";
			mes "我们这里有位医术极高明的大夫，";

			mes "[士兵]";
			mes "据说没有她治不好的病。";
			mes "当然，是真是假我可不敢打包票。";
			viewpoint 1,263,94,3,0xCE6300;
			break;
		case 3:
			mes "[士兵]";
			mes "官厅是联邦政府办公的地方，";
			mes "位于 ^A5BAAD+^000000 处。";
			next;
			mes "[士兵]";
			mes "如果遇到什么麻烦，";
			mes "可以去找官厅的官员谈一谈。";
			viewpoint 1,309,80,4,0xA5BAAD;
			break;
		case 4:
			mes "[士兵]";
			mes "武器店位于 ^55FF33+^000000 处。";
			next;
			mes "[士兵]";
			mes "那里陈列着洛阳的老师傅们";
			mes "打造出的精妙武器，";
			mes "定会让你大开眼界。";
			viewpoint 1,145,174,5,0x55FF33;
			break;
		case 5:
			mes "[士兵]";
			mes "道具店位于 ^3355FF+^000000 处。";
			next;
			mes "[士兵]";
			mes "知己知彼，百战不殆！";
			mes "未雨绸缪总好过事后后悔，";
			mes "不妨去瞧瞧他们备的货吧？";
			viewpoint 1,135,98,6,0x3355FF;
			break;
		case 6:
			mes "[士兵]";
			mes "旅途劳顿时，建议你去酒馆坐坐，";
			mes "位置就在 ^00FF00+^000000 处。";
			next;
			mes "[士兵]";
			mes "酒馆里既能结识各地游客，";
			mes "也能打听到有用的消息。";
			viewpoint 1,280,167,7,0x00FF00;
			break;
		case 7:
			mes "[士兵]";
			mes "如果想清除小地图上的所有标记，";
			mes "请选择菜单中的''删除小地图上的所有标记。''";
		}
		break;
	case 2:
		viewpoint 2,218,255,2,0xFF00FF;
		viewpoint 2,263,94,3,0xFF00FF;
		viewpoint 2,309,80,4,0xFF00FF;
		viewpoint 2,145,174,5,0xFF00FF;
		viewpoint 2,135,98,6,0xFF00FF;
		viewpoint 2,280,167,7,0xFF00FF;
		mes "[士兵]";
		mes "好了，小地图上的标记都已清除。";
		mes "想了解建筑位置时，随时来问我。";
		break;
	case 3:
		mes "[士兵]";
		mes "偶尔独自探索";
		mes "也别有一番乐趣。";
		mes "路上小心。";
	}
	close;
}
