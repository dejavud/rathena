morocc,153,286,6	script	解说员#moc::MocGuide	707,{
	cutin "moc_soldier",2;
	mes "[梦罗克解说员]";
	mes "欢迎来到卢恩·米德加兹王国的";
	mes "边境城镇梦罗克。";
	mes "如果在城里找不到地方，";
	mes "请随时向我求助。";
	set .@loop1,1;
	while (.@loop1) {
		next;
		switch(select("城市导览:删除小地图上的标记:注意事项:取消")) {
		case 1:
			mes "[梦罗克解说员]";
			mes "请从菜单中选择您想去的地点。";
			if (.@compass_check == 0) {
				mes "需要我把这些地点";
				mes "标记到您的小地图上吗？";
				next;
				if (select("是。:否。") == 1) set .@compass_check,1;
			}
			set .@loop2,1;
			while(.@loop2) {
				if (.@wait_button_chk == 0) set .@wait_button_chk,1;
				else next;

				switch(select("^FF0000盗贼公会^000000:武器店:旅馆:酒馆:佣兵公会:锻造坊:取消")) {
				case 1:
					mes "[梦罗克解说员]";
					mes "盗贼公会负责盗贼职业的转职。";
					mes "听说他们就在附近的金字塔里面……";
					if (.@compass_check)
						viewpoint 1,24,297,2,0xFF0000;
					break;
				case 2:
					mes "[梦罗克解说员]";
					mes "武器店在梦罗克的东南端。";
					if (.@compass_check)
						viewpoint 1,253,56,3,0xFF00FF;
					break;
				case 3:
					mes "[梦罗克解说员]";
					mes "梦罗克的东南端和东北端";
					mes "各有一家旅馆可以休息。";
					if (.@compass_check) {
						viewpoint 1,197,66,4,0xFF00FF;
						viewpoint 1,273,269,5,0xFF00FF;
					}
					break;
				case 4:
					mes "[梦罗克解说员]";
					mes "酒馆在梦罗克的东北方向。";
					if (.@compass_check)
						viewpoint 1,52,259,6,0xFF00FF;
					break;
				case 5:
					mes "[梦罗克解说员]";
					mes "佣兵公会位于梦罗克东部。";
					if (.@compass_check)
						viewpoint 1,284,171,7,0x00FF00;
					break;
				case 6:
					mes "[梦罗克解说员]";
					mes "锻造坊在梦罗克中心";
					mes "稍微偏西南的位置。";
					if (.@compass_check)
						viewpoint 1,47,47,7,0xFF00FF;
					break;
				case 7:
					mes "[梦罗克解说员]";
					mes "如果不想让小地图继续显示位置标记，";
					mes "请选择''删除小地图上的标记''。";
					set .@loop2,0;
				}
			}
			break;
		case 2:
			viewpoint 2,237,41,2,0x00FF00;
			viewpoint 2,237,41,3,0x0000FF;
			viewpoint 2,46,345,4,0x00FF00;
			viewpoint 2,175,220,5,0xFF0000;
			viewpoint 2,175,220,6,0xFF0000;
			viewpoint 2,175,220,7,0xFF0000;
			set .@compass_check,0;
			break;
		case 3:
			mes "[梦罗克解说员]";
			mes "借助魔法与技术的进步，";
			mes "我们的导览系统得到升级，";
			mes "可以把地点标记到您的小地图上，";
			mes "方便您在城市里认路。";
			next;
			mes "[梦罗克解说员]";
			mes "小地图在屏幕右上角。";
			mes "如果看不到，请按 Ctrl + Tab，";
			mes "或点击基本信息窗口里的''地图''按钮。";
			next;
			mes "[梦罗克解说员]";
			mes "点击小地图上的''+''和''-''";
			mes "可以放大或缩小显示范围。";
			mes "祝您在梦罗克旅途愉快。";
			break;
		case 4:
			mes "[梦罗克解说员]";
			mes "好的，冒险者，";
			mes "在外面可别惹太多麻烦哦。";
			close2;
			set .@loop1,0;
		}
	}
	cutin "moc_soldier",255;
	end;
}

morocc,54,97,0	duplicate(MocGuide)	解说员#2moc	707
