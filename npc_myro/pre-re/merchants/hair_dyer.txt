prt_in,243,168,4	script	造型师	91,{
	mes "[造型师]";
	mes "欢迎~！";
	mes "有什么可以帮您？";
	next;
	mes "[造型师]";
	if (Sex == SEX_MALE) {
		mes "哦，不行！您的头发受损了。";
		mes "看来您需要专业护理。请坐这边，来。";
	}
	else {
		if (rand(20) > 11) {
			mes "诶！？哎呀！哦不不不不！";
			mes "您的头发受损得好严重！";
			mes "这样放任不管可不行。";
			next;
			mes "[造型师]";
			mes "让我来护理您的头发好吗？拜托？";
		}
		else mes "哇！您的头发只要染个色就完美了~";
		mes "换个形象染个发怎么样？";
	}
	next;
	while(1) {
		switch(select("染发:小贴士与信息:取消")) {
		case 1:
			mes "[造型师]";
			mes "好好，选得好~";
			mes "这是您改头换面、绽放美丽的机会！";
			mes "请选您喜欢的颜色。";
			next;
			while(1) {
				if (.@choose_success == 1) {
					mes "[造型师]";
					mes "觉得怎么样？想换个别的颜色吗？";
					next;
					if (select("是:否") == 1) {
						mes "[造型师]";
						mes "好！请选您喜欢的颜色。";
						next;
					}
					else  {
						mes "[造型师]";
						mes "嗯，我有点失望，本想做得更好。";
						mes "不过我保证下次会更出色。请再来~";
						close;
					}
				}

				while(1) {
					switch(select("红色，谢谢。:黄色，谢谢。:紫色，谢谢。:赤色，谢谢。:绿色，谢谢。:蓝色，谢谢。:白色，谢谢。:黑色，谢谢。:其实，我喜欢现在的头发。")) {
					case 1: set .@headpalette,8; break;
					case 2: set .@headpalette,1; break;
					case 3: set .@headpalette,2; break;
					case 4: set .@headpalette,3; break;
					case 5: set .@headpalette,4; break;
					case 6: set .@headpalette,5; break;
					case 7: set .@headpalette,6; break;
					case 8: set .@headpalette,7; break;
					case 9:
						if (.@choose_success != 0) {
							mes "[造型师]";
							mes "您一定很喜欢现在的发色~";
							close;
						}
						mes "[造型师]";
						mes "诶~？您不染发吗？我有点难过……";
						close;
					}

					if (.@headpalette == getlook(VAR_HEADPALETTE)) {
						mes "[造型师]";
						mes "诶？可这就是您现在的发色啊。";
						mes "请选一个别的颜色。";
						next;
					}
					else {
						switch(.@headpalette) {
						case 1: callsub S_NoDye,976,"黄色"; break;
						case 2: callsub S_NoDye,978,"紫色"; break;
						case 3: callsub S_NoDye,980,"赤色"; break;
						case 4: callsub S_NoDye,979,"绿色"; break;
						case 5: callsub S_NoDye,981,"蓝色"; break;
						case 6: callsub S_NoDye,982,"白色"; break;
						case 7: callsub S_NoDye,983,"黑色"; break;
						case 8: callsub S_NoDye,975,"红色"; break;
						}
						if (Zeny < 1000) {
							mes "[造型师]";
							mes "费用是 1000 Zeny。";
							mes "您的钱不够吗……？";
							close;
						}

						switch(.@headpalette) {
						case 1: delitem 976,1; break; //Lemon_Dyestuffs
						case 2: delitem 978,1; break; //Violet_Dyestuffs
						case 3: delitem 980,1; break; //Orange_Dyestuffs
						case 4: delitem 979,1; break; //Darkgreen_Dyestuffs
						case 5: delitem 981,1; break; //Cobaltblue_Dyestuffs
						case 6: delitem 982,1; break; //White_Dyestuffs
						case 7: delitem 983,1; break; //Black_Dyestuffs
						case 8: delitem 975,1; break; //Scarlet_Dyestuffs
						}
						set Zeny, Zeny-1000;
						setlook VAR_HEADPALETTE,.@headpalette;
						set .@choose_success,1;
						break;
					}

				}

			}

		case 2:
			mes "[造型师]";
			mes "心情低落的时候、失恋的时候、";
			mes "想给谁留下好印象的时候，";
			mes "甚至只是想引人注目的时候……";
			next;
			mes "[造型师]";
			mes "在那些特别的场合和时刻，";
			mes "谁不想拥有一个属于自己的发型呢？";
			mes "只要您带齐对应的染料，";
			mes "我就能让您的头发焕发光彩。";
			next;
			mes "[造型师]";
			mes "也别太担心费用。";
			mes "追求美和追求一切美好事物一样。";
			mes "呵呵~";
			next;
			mes "[造型师]";
			mes "* 费用与信息 *";
			mes "- 1 个您想染的发色对应的染料。";
			mes "- 1000 Zeny 费用。";
			close;

		case 3:
			mes "[造型师]";
			mes "无论男女……";
			mes "每个人都有变美的权利和义务。";
			close;
		}

	}

S_NoDye:
	if (countitem(getarg(0)) == 0) {
		mes "[造型师]";
		mes "诶？！可是要把头发染成"+getarg(1)+"，需要 '"+getitemname(getarg(0))+"' 才行啊……";
		close;
	}
	return;
}
