que_ng,187,149,3	script	子弹盒商人 肯尼::mdk	4_M_01,{
	mes "[肯尼]";
	if (BaseClass == Job_Gunslinger) {
		mes "欢迎来到我的子弹盒商店。";
		mes "你也知道，子弹数量多的时候，";
		mes "装进子弹盒里携带起来更方便。";
		mes "那么，你想要点什么？";
		next;
		switch(select("雷击子弹盒:暗盲子弹盒:毒物子弹盒:冰冻子弹盒:火焰子弹盒:弹药桶:红色弹药桶:银色弹药桶:取消")) {
		case 1: callsub S_Casing,13204,12144;// <bullet id>, <pack id>
		case 2: callsub S_Casing,13206,12145;
		case 3: callsub S_Casing,13205,12146;
		case 4: callsub S_Casing,13207,12147;
		case 5: callsub S_Casing,13203,12148;
		case 6: callsub S_Casing,13200,12149;
		case 7: callsub S_Casing,13202,12150;
		case 8: callsub S_Casing,13201,12151;
		case 9:
			mes "[肯尼]";
			mes "好吧，要是你觉得子弹多得没处放了，";
			mes "就来找我吧。";
			mes "用我的子弹盒来存放子弹，是个聪明的主意。";
			close;
		}
	}
	mes "欢迎来到我的商店。";
	mes "我在这里为神枪手们提供子弹盒和弹药桶。";
	mes "抱歉，看来我的服务对你没什么用处，";
	mes "冒险者。";
	next;
	mes "[肯尼]";
	mes "哎，不过要是你刚好认识什么神枪手，";
	mes "就介绍他们来我这里吧。";
	mes "子弹这东西，永远不嫌多。";
	close;

S_Casing:
	.@bullet_id = getarg(0);
	.@pack_id = getarg(1);

	mes "[肯尼]";
	mes "那么，你可以用";
	mes callfunc("F_InsertPlural",500,getitemname(.@bullet_id));
	mes "加 500 Zeny 换 1 个 "+getitemname(.@pack_id)+"。";
	mes "先确认你身上的子弹和 Zeny";
	mes "够不够这次交换。";
	next;
	mes "[肯尼]";
	mes "用 "+callfunc("F_InsertPlural",500,getitemname(.@bullet_id))+" 加 500 Zeny";
	mes "可以换 1 个 "+getitemname(.@pack_id)+"。";
	next;
	mes "[肯尼]";
	mes "记住了，一次最多只能给你 50 个。";
	mes "请输入你想要领取的数量。";
	mes "想取消的话，只要输入 ''0'' 就行。";
	next;
	input .@amount;
	mes "[肯尼]";
	if (.@amount > 50 || .@amount == 0) {
		mes "嘿，那么多我可给不了你。";
		mes "请再试一次，输入 50 以内的数字。";
	}
	else if (countitem(.@bullet_id) >= (500*.@amount)) {
		if (Zeny >= (500*.@amount)) {
			if (checkweight(.@pack_id,.@amount) == 0) {
				mes "嘿，你的物品栏里塞了太多破烂了。";
				mes "先腾出一些空间，再来把子弹换成子弹盒吧，";
				mes "好吗？";
			} else {
				mes "好，这些就是你的子弹盒了。";
				mes "谢谢光顾我的商店，";
				mes "希望你把每一发弹药都用在刀刃上。";
				Zeny = Zeny -500*.@amount;
				delitem .@bullet_id,(500*.@amount);
				getitem .@pack_id,.@amount;
			}
		} else {
			mes "抱歉，你的 Zeny 不够换这些子弹盒。";
			mes "攒够了钱，再来我的商店吧。";
		}
	} else {
		mes "抱歉，你的子弹不够换这些子弹盒。";
		mes "还是再确认一遍物品栏，然后再来找我吧。";
	}
	close;
}
