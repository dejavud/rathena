alberta,118,160,3	script	子弹袋商人 托尼#alb::bdt	4_M_04,{
	mes "[托尼]";
	if (BaseClass == Job_Gunslinger) {
		mes "我是托尼，子弹袋商人。";
		mes "弹药不够的时候，随时来找我。";
		mes "只要把材料带来，我就给你做出需要的子弹。";
		next;
		switch(select("毒物子弹:火焰子弹:雷击子弹:暗盲子弹:冰冻子弹:取消")) {
		case 1: callsub S_BulletTrade,937,10,13205;// <item required>, <amount req>, <bullet id>
		case 2: callsub S_BulletTrade,7097,2,13203;
		case 3: callsub S_BulletTrade,7053,3,13204;
		case 4: callsub S_BulletTrade,1024,5,13206;
		case 5: callsub S_BulletTrade,7054,2,13207;
		case 6:
			mes "[托尼]";
			mes "改变主意了？";
			mes "好吧，需要子弹的时候，我随时都在这里。";
			mes "觉得弹药不够用了就再来吧，神枪手。";
			close;
		}
	}
	mes "嘿，我是托尼。";
	mes "我负责为神枪手们分发和制作子弹。";
	mes "我们公会就是喜欢这么做事。";
	next;
	mes "[托尼]";
	mes "如果你是来买子弹的，那可抱歉了。";
	mes "我只跟正式的神枪手做生意。";
	close;

S_BulletTrade:
	.@required_id = getarg(0);
	.@required_amount = getarg(1);
	.@bullet_id = getarg(2);

	mes "[托尼]";
	mes "每 30 发";
	mes callfunc("F_InsertPlural",30,getitemname(.@bullet_id))+",";
	mes "你要给我 1 个强化武器金属-级数一、";
	mes "1 个强化武器金属-级数二，还有";
	mes callfunc("F_InsertPlural",.@required_amount,getitemname(.@required_id))+".";
	next;
	mes "[托尼]";
	mes "记住了，一次最多只能给你 500 组，";
	mes "每组 30 发子弹。";
	mes "请输入你想要的组数。想取消的话，";
	mes "只要输入 ''0'' 就行。";
	next;
	input .@amount;
	mes "[托尼]";
	if (.@amount < 1 || .@amount > 500) {
		mes "嘿，那么多子弹我可给不了你。";
		mes "想用物品换子弹的话，";
		mes "别忘了输入 500 以内的数字。";
	}
	else if (countitem(1010) >= .@amount && countitem(1011) >= .@amount && countitem(.@required_id) >= (.@amount*.@required_amount)) {
		if (checkweight(.@bullet_id,.@amount * 30) == 0) {
			mes "咦？你的物品栏没有那么多空间放这些子弹。";
			mes "先腾出一些空间再来吧。";
			mes "可以把一些东西存进卡普拉仓库。";
		} else {
			mes "好，一切都齐了。我收下这些材料，";
			mes "这些就是你的子弹了。";
			mes "跟你做生意真是愉快~";
			delitem 1010,.@amount; //Phracon
			delitem 1011,.@amount; //Emveretarcon
			delitem .@required_id,.@amount * .@required_amount;
			getitem .@bullet_id,.@amount * 30;
		}
	} else {
		mes "嗯。看来你的材料不够做那么多子弹。";
		mes "好吧，没关系。";
		mes "把需要的东西都凑齐了再来，";
		mes "好吗？";
	}
	close;
}
