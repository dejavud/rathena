//callfunc "F_CoolEventCorp","<menu>","<location>","<save map>",<x>,<y>,"<warp map>";
function	script	F_CoolEventCorp	{
	cutin "zonda_01",2;
	mes "[准它企划社社员]";
	mes "欢迎光临准它企划社。";
	mes "我们的员工始终致力于提供超越";
	mes "您期望的优质服务。";
	mes "那么,今天有什么可以帮您?";
	next;
	switch(select(getarg(0))) {
	case 1:
		mes "[准它企划社社员]";
		mes "您的重生点已保存于此";
		mes getarg(1)+"。";
		mes "感谢您使用准它企划社的服务~";
		savepoint getarg(2),getarg(3),getarg(4),1,1;
		close2;
		break;
	case 2:
		mes "[准它企划社社员]";
		if(!callfunc("F_CanOpenStorage")){
			mes "非常抱歉,";
			mes "使用仓库服务需要初心者的基本";
			mes "技能达到等级6。";
			close2;
			break;
		}
		if(Zeny < 40){
			mes "非常抱歉,";
			mes "您的Zeny不够支付仓库服务费用。";
			mes "使用仓库的费用是 40 Zeny。";
			close2;
			break;
		}
		set Zeny, Zeny - 40;
		mes "我马上为您打开个人仓库。";
		mes "感谢您使用准它企划社的服务,";
		mes "祝您愉快~";
		callfunc("F_CheckKafCode");	//check your storage password, if set
		close2;
		openstorage;
		break;
	case 3:
		mes "请选择您的目的地。";
		next;
		if (VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE))
			set .@cost,4400;
		else
			set .@cost,2200;
		.@destination$ = getarg(5);
		if (select(.@destination$ +" -> "+ .@cost +" z:取消") == 1) {
			if ((countitem(7060) != 0) || (Zeny >= .@cost)) {
				if (countitem(7060) != 0) {
					delitem 7060,1; //Warp_Free_Ticket
				}
				else {
					set Zeny,Zeny-.@cost;
				}
				if (.@destination$ == "伯仁斯") warp "veins",205,101;
				else if (.@destination$ == "拉赫") warp "rachel",115,125;
				cutin "",255;
				end;
			}
			else {
				mes "[准它企划社社员]";
				mes "非常抱歉,";
				mes "您的zeny不够支付传送服务费用。";
				mes "传送到";
				mes ""+ .@destination$ +" 的费用是 "+.@cost+" zeny。";
			}
		}
		close2;
		break;
	case 4:
		mes "[准它企划社社员]";
		if(BaseClass != Job_Merchant){
			mes "非常抱歉,手推车租赁服务";
			if (checkre(0))
				mes "仅限商人系职业使用。";
			else {
				mes "仅限商人、铁匠、神匠、";
				mes "炼金术士和创造者使用。";
			}
			close2;
			break;
		}
		if(getskilllv("MC_PUSHCART") == 0){
			mes "学会手推车技能后才能租用手推";
			mes "车。";
			close2;
			break;
		}
		if(checkcart() == 1){
			mes "您已经租了一辆手推车。";
			mes "非常抱歉,";
			mes "每位顾客同一时间只能租用一辆。";
			close2;
			break;
		}
		if(countitem(7061)){
			delitem 7061,1; //Cart_Free_Ticket
			setcart;
			close2;
			break;
		}
		mes "手推车的租金是 800 Zeny。";
		mes "您要租一辆手推车吗?";
		next;
		if(select("租用手推车:取消") == 1){
			if(Zeny < 800){
				mes "[准它企划社社员]";
				mes "非常抱歉,";
				mes "您的Zeny不够支付 800 Zeny 的";
				mes "手推车租金。";
			} else {
				set Zeny, Zeny - 800;
				setcart;
			}
		}
		close2;
		break;
	case 5:
		callfunc("F_SetKafCode","[准它企划社社员]","准它企划社");
		//the function closes msg and removes cutins
	case 6:
	default:
		mes "[准它企划社社员]";
		mes "准它企划社始终致力于为顾客提";
		mes "供最好的服务。";
		mes "请提出您的意见和真诚的反馈,";
		mes "帮助我们做到最好。";
		mes "谢谢。";
		close2;
	}
	cutin "", 255;
	end;
}
