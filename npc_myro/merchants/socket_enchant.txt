function	script	Func_Socket	{
	.@item_id = getarg(0);
	.@zeny_req = getarg(4);
	setarray .@item_req[0], getarg(5), getarg(7,0);
	setarray .@req_amount[0], getarg(6), getarg(8,0);

	disable_items;
	mes "[世亚布连]";
	mes "想给 "+ getitemname(.@item_id) +" 打洞吗？";
	mes "好的，请带来 ^FF0000"+ .@req_amount[0] +" 个"+ getitemname(.@item_req[0]) +"^000000、"+ (.@item_req[1] != 0 && .@req_amount[1] != 0 ? ("^FF0000"+ .@req_amount[1] +" 个"+ getitemname(.@item_req[1]) +"^000000、") : "") +"以及 ^FF0000" + (.@zeny_req >= 1000 ? (.@zeny_req /1000 +",000") : .@zeny_req) +",000 Zeny^000000 的手续费。";
	mes "啊，别忘了把那件 "+ getitemname(.@item_id) +" 也带来！";
	next;
	mes "[世亚布连]";
	if (getiteminfo(.@item_id, ITEMINFO_LOCATIONS) & EQP_HAND_R) // weapon
	{
		mes "只要材料和 Zeny 齐了，我现在就可以试着打洞。";
		mes "不过你得知道，我是有可能失败的。";
		mes "所以，我得先给你一个忠告……";
		next;
		mes "[世亚布连]";
		mes "如果这次给武器打洞失败，^FF0000武器^000000和^FF0000插在上面的卡片都会被摧毁^000000。";
	}
	else // armor
	{
		mes "只要材料、手续费和防具都齐了，我们就可以开始打洞。";
		mes "但在那之前，我必须提醒你其中的风险。";
		next;
		mes "[世亚布连]";
		mes "如果这次给防具打洞失败，^FF0000防具^000000、^FF0000它的精炼等级^000000和^FF0000插在上面的卡片都会被摧毁^000000。";
	}
	mes "你还想试着打洞吗？";
	next;
	switch(select("尝试打洞:取消"))
	{
		case 1:
			if ((Zeny >= .@zeny_req*1000) && countitem(.@item_req[0]) >= .@req_amount[0] && countitem(getarg(7,512)) >= .@req_amount[1] && countitem(.@item_id) > 0)// if there's no getarg(7) and getarg(8) provided, check itemcount against 0 (we still need a valid item ID though)
			{
				mes "[世亚布连]";
				mes "那么，开工了！";
				mes "你最好祈祷能成功。";
				next;
				set .@a, rand(1,100);
				if ((.@a > getarg(2)) && (.@a < getarg(3)))
				{
					specialeffect getarg(3) == 51 ? EF_LORD : EF_SANCTUARY; // only S Class items have Lord of Vermillion effect
					mes "[世亚布连]";
					mes "太好了，看来成功了。";
					mes "做得相当漂亮。恭喜你！";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					getitem getarg(1),1;
					next;
					mes "[世亚布连]";
					mes "再见啦，朋友！";
					close;
				}
				else
				{
					specialeffect EF_SUI_EXPLOSION;
					mes "[世亚布连]";
					mes "哇！……非常抱歉，失败了。";
					mes "不过这可不是我的错。";
					mes "这是运气问题，是上天注定的，懂吗？";
					mes "别太沮丧，下次再试吧。";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					next;
					mes "[世亚布连]";
					mes "祝你下次好运！";
					close;
				}
			}
			else
			{
				mes "[世亚布连]";
				mes "我很想帮你打洞，但你还缺了些东西。";
				mes "装备、材料和 Zeny 都带齐了吗？";
				close;
			}
		case 2:
			mes "[世亚布连]";
			mes "需要时间考虑一下，是吧？";
			mes "好吧，我能理解。";
			mes "不过记住，总是求稳的话，人生可就没意思了~";
			close;
	}
}
