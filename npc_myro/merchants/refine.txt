//============================================================
//= Main Refiner Function
//============================================================
//= To allow auto safe refining/multiple refining set the
//= second argument to '1' in the function call.
//= If you enable this function, be sure to edit the value of
//= .@safe to the max safe refine in refine.yml as well.
//============================================================
function	script	refinemain	{
	disable_items;
	.@npc_name$ = getarg(0);
	.@features = getarg(1);
	mes "["+ .@npc_name$ +"]";
	mes "我是铁匠。";
	mes "各种武器、防具和装备我都能精炼，";
	mes "告诉我你想精炼什么吧。";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(.@i = 1; .@i<getarraysize(.@indices); ++.@i) {
		if(getequipisequiped(.@indices[.@i])) {
			.@menu$ = .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			.@equipped = 1;
		}
		.@menu$ = .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "你身上好像没有我能精炼的物品……";
		close;
	}
	.@part = .@indices[select(.@menu$)];

	if(!getequipisequiped(.@part)) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "那个部位你什么都没穿，";
		mes "没有我能精炼的东西。";
		emotion ET_FRET;
		close;
	}
	//Check if the item is refinable...
	if(!getequipisenableref(.@part)) {
		mes "["+ .@npc_name$ +"]";
		mes "这件物品好像";
		mes "根本没法精炼……";
		close;
	}
	//Check to see if the items is already +10
	if(getequiprefinerycnt(.@part) >= 10) {
		mes "["+ .@npc_name$ +"]";
		mes "这件不能再精炼了。";
		mes "已经到顶了！";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );

	if( .@itemtype == IT_ARMOR ){
		.@equip_lv = getequiparmorlv( .@part );

		switch( .@equip_lv ){
			case 1:
				.@safe = 4;
				break;
			default:
				// TODO:
				close;
		}

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if( VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE) ){
			switch( .@equip_lv ){
				case 1:
					.@price = .@price * 10;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else if( .@itemtype == IT_WEAPON ){
		.@equip_lv = getequipweaponlv( .@part );

		switch( .@equip_lv ){
			case 1:
				.@safe = 7;
				break;
			case 2:
				.@safe = 6;
				break;
			case 3:
				.@safe = 5;
				break;
			case 4:
				.@safe = 4;
				break;
			default:
				// TODO:
				close;
		}

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if( VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE) ){
			switch( .@equip_lv ){
				case 1:
					.@price = .@price * 40;
					break;
				case 2:
					.@price = .@price * 50;
					break;
				case 3:
					.@price = .@price * 2;
					break;
				case 4:
					.@price = .@price * 2;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else{
		.@safe = 4;
	}

	// TODO: What was this? Maybe shadow?
	// case 5: .@price = .@price * 10; break;

	if(.@features != 1) {
		mes "["+ .@npc_name$ +"]";
		mes "精炼这个需要";
		mes "1 个^003366"+getitemname(.@material)+"^000000，";
		mes "手续费是 " + .@price + " Zeny。";
		mes "真的要继续吗？";
		next;
		if(select("是:否") == 2){
			mes "["+ .@npc_name$ +"]";
			mes "也是……";
			mes "不用着急。";
			mes "慢慢考虑吧。";
			close;
		}
		if(getequippercentrefinery(.@part) < 100) {
			mes "["+ .@npc_name$ +"]";
			mes "糟糕！再继续精炼的话，";
			mes "这件装备有可能";
			switch(.@material) {
			case 985:
				mes "会损坏！";
				mes "也就是说，^FF0000这件防具^000000和^FF0000插在上面的卡片^000000、";
				mes "附加的特殊属性，^FF0000都会消失^000000。";
				break;
			default:
				mes "会损坏，你会^FF0000失去这件武器^000000、";
				mes "^FF0000插在武器上的卡片^000000和附加的特殊属性。";
				break;
			}
			next;
			mes "["+getarg(0)+"]";
			mes "我已经说得够清楚了。";
			mes "武器一旦损坏，";
			mes "就再也找不回来了。";
			mes "你真的有可能";
			mes "^FF0000永远失去这件武器^000000。";
			mes "还要继续精炼吗？";
			next;
			if(select("是:否") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "我完全同意……";
				mes "我虽然是个出色的精炼师，但有时也会失手。";
				close;
			}
		}
		if((countitem(.@material) < 1) || (Zeny < .@price)) {
			mes "["+ .@npc_name$ +"]";
			mes "你的 Zeny 或"+getitemname(.@material)+"";
			mes "好像不够……";
			mes "再去弄一些吧。";
			mes "我今天";
			mes "一直都在这里。";
			close;
		}
		Zeny = Zeny-.@price;
		delitem .@material,1;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
			mes "["+ .@npc_name$ +"]";
			emotion ET_FRET;
			mes "等一下……";
			mes "你当我是傻子吗？！";
			mes "你趁我不注意把装备换了！";
			mes "给我出去！";
			close;
		}

		if(getequippercentrefinery(.@part) <= rand(100)) {
			failedrefitem .@part;
			mes "["+ .@npc_name$ +"]";
			emotion (!rand(5))?ET_MONEY:ET_HUK;
			.@lose = rand(1,3);
			if (.@lose == 1) {
				mes "天哪！";
				mes "该死！又失败了！";
				mes "实在太抱歉了，不过你知道，熟能生巧嘛。";
				mes "嗯，是吧？";
				mes "嘿嘿……";
			} else if(.@lose == 2) {
				mes "不要啊！";
				mes "碎了！";
				mes "对……对不起！";
			} else {
				mes "糟糕！";
				mes "它经受不住";
				mes "更多的锤炼了！";
				mes "真是抱歉……";
			}
			close;
		}
		mes "["+getarg(0)+"]";
		successrefitem .@part;
		emotion ET_SMILE;
		.@win = rand(1,3);
		if (.@win == 1) {
			mes "完美！";
			mes "嘿嘿！";
			mes "再一次，";
			mes "大师出手，";
			mes "毫无瑕疵~";
		} else if(.@win == 2) {
			mes "成功了……！";
			mes "我这惊人的手艺，";
			mes "今天依旧";
			mes "光彩夺目。";
		} else {
			mes "嘿嘿！";
			mes "完成了。";
			mes "毫无疑问，";
			mes "我的手艺包你满意。";
			mes "纯粹、彻底的完美~";
		}
		close;
	}

// New Refining Functions ========================
	if (.@refinerycnt < .@safe) {
		mes "["+ .@npc_name$ +"]";
		mes "我可以精炼到安全上限，";
		mes "也可以精炼到你指定的次数。";
		mes "你选吧。";
		next;
		.@menu2 = select("精炼到安全上限。","我自己决定次数。","我改变主意了……");
	} else
		.@menu2 = 2;
	switch(.@menu2){
	case 1: 
		.@refinecnt = .@safe - .@refinerycnt;
		break;
	case 2:
		next;
		mes "["+ .@npc_name$ +"]";
		mes "想让我精炼几次？";
		next;
		input .@refinecnt;
		.@refinecheck = .@refinecnt + .@refinerycnt;
		if (.@refinecnt < 1 || .@refinecheck > 10) {
			mes "["+ .@npc_name$ +"]";
			mes "这件装备不能精炼那么多次。";
			close;
		}
		if(.@refinecheck > .@safe) {
			.@refinecheck = .@refinecheck - .@safe;
			mes "["+ .@npc_name$ +"]";
			mes "这样的话，装备会超出安全上限再精炼 " + .@refinecheck + " 次。";
			mes "装备可能会被摧毁……可以吗？";
			next;
			if(select("可以……","不行……") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "既然你这么说……那就这样吧。";
				close;
			}
		}
		break;
	case 3:
		next;
		mes "["+ .@npc_name$ +"]";
		mes "既然你这么说……那就这样吧。";
		close;
	}
	.@fullprice = .@price * .@refinecnt;
	mes "["+ .@npc_name$ +"]";
	mes "那样需要 " + .@refinecnt + " 个" + getitemname(.@material) + "和 " + .@fullprice + " Zeny。可以吗？";
	next;
	if(select("可以","不行……") == 2){
		mes "["+ .@npc_name$ +"]";
		mes "既然你这么说……那就这样吧。";
		close;
	}
	if(countitem(.@material) < .@refinecnt || Zeny < .@fullprice) {
		mes "["+ .@npc_name$ +"]";
		mes "你就只有这些吗？";
		mes "可惜我不能降价干活。";
		mes "你也替我想想吧。";
		close;
	}
	Zeny = Zeny - .@fullprice;
	delitem .@material,.@refinecnt;
	while(.@refinecnt){
		if (getequipisequiped(.@part) == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "喂……你身上什么都没穿啊……";
			close;
		}
		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
				callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt) || (.@menu2 == 1 && getequippercentrefinery(.@part) < 100)) {
			mes "["+ .@npc_name$ +"]";
			mes "铛……不对，你当我是傻子吗？！";
			mes "你把装备换了……";
			mes "快给我出去，不然我用锤子砸晕你！！";
			close;
		} 
		mes "铛、铛！！！";
		if(.@menu2 == 2 && getequippercentrefinery(.@part) <= rand(100)) {
			failedrefitem .@part;
			emotion ET_HUK;
			mes "["+ .@npc_name$ +"]";
			mes "哇啊！！！";
			mes "太抱歉了……我提醒过你可能会这样的……";
			.@refinecnt = .@refinecnt - 1;
			if(.@refinecnt == 0) close;
			mes "把没用掉的 Zeny 和材料还给你……";
			getitem .@material,.@refinecnt;
			.@fullprice = .@refinecnt * .@price;
			Zeny = Zeny + .@fullprice;
			close;
		}
		successrefitem .@part;
		emotion ET_BEST;
		.@refinecnt = .@refinecnt - 1;
		.@refinerycnt = getequiprefinerycnt(.@part);
		next;
	}
	mes "["+ .@npc_name$ +"]";
	mes "全部完成了……欢迎再来。";
	close;
}

function	script	phramain	{
	if (checkweight(1201,1) == 0) {
		mes "- 等一下！！ -";
		mes "- 你现在携带的 -";
		mes "- 物品太多了。 -";
		mes "- 请减轻一些重量 -";
		mes "- 之后再来试试吧。 -";
		close;
	}
	.@npc_name$ = getarg(0);
	mes "["+ .@npc_name$ +"]";
	mes "我卖两种锻造武器";
	mes "用的金属。";
	mes "1 级武器用的^007777强化武器金属-级数一^000000，";
	mes "还有 2 级武器用的";
	mes "^007777强化武器金属-级数二^000000。";
	next;
	switch(select("强化武器金属-级数一 - 200 Zeny:强化武器金属-级数二 - 1000 Zeny:问问其他金属")) {
	case 1:
		.@material = 1010;
		.@price = 200;
		break;
	case 2:
		.@material = 1011;
		.@price = 1000;
		break;
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "其他金属？";
		mes "嗯，强化更高级的武器或任何防具，";
		mes "需要特殊的金属。";
		mes "不过你知道的，神之金属和铝这东西，实在是";
		mes "很难找到……";
		close;
	}
	mes "["+ .@npc_name$ +"]";
	mes "那么要买几个？";
	mes "不想要的话，请输入数字''0''。";
	next;
	while(1) {
		input .@input;
		if (.@input == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "交易";
			mes "已取消。";
			close;
		}
		else if (.@input < 0 || .@input > 500) {
			mes "["+ .@npc_name$ +"]";
			mes "好吧，最多";
			mes "可以买 500 个。";
			mes "不能再多了，";
			mes "明白吗？";
			mes "很好。";
			next;
		}
		else {
			break;
		}
	}
	.@sell = .@input * .@price;
	if (Zeny < .@sell) {
		mes "["+ .@npc_name$ +"]";
		mes "呃……";
		mes "买 "+ .@input +" 个的话，";
		mes "你的 Zeny 不够。";
		close;
	}
	if (checkweight(.@material,.@input) == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "嗯……";
		mes "背包空间不够，我可没法给你东西。";
		mes "何不把多余的东西存进卡普拉仓库再来？";
		close;
	}
	getitem .@material,.@input;
	Zeny = Zeny-.@sell;
	mes "["+ .@npc_name$ +"]";
	mes "给你！";
	mes "感谢惠顾。";
	close;
}

// Equipment Repair Function
//============================================================
function	script	repairmain	{
	.@repairprice = 5000;
	.@npc_name$ = getarg(0);
	mes "["+ .@npc_name$ +"]";
	mes "你好啊！";
	mes "有物品需要";
	mes "修理吗？";
	mes "修理物品找我，";
	mes "准没错！";
	next;
	switch(select("其实，我有些物品……:现在没有。")) {
	case 1:
		.@checkitem = 1;
		while (1) {
			if (getbrokenid(.@checkitem) == 0) {
				break;
			}
			.@checkitem = .@checkitem+1;
		}
		.@checkitem = .@checkitem-1;
		if (!.@checkitem) {
			mes "["+ .@npc_name$ +"]";
			mes "哦，天哪，真了不起！";
			mes "你一定很爱惜物品。";
			mes "你的物品一件都没损坏！";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "要是人人都像你这样，我就要失业了！！";
			mes "哈哈~！";
			close;
		}
		mes "["+ .@npc_name$ +"]";
		mes "嗯……";
		mes "让我看看……";
		mes "你所有的物品中，";
		mes "有 " + .@checkitem + " 件损坏了。";
		mes "要修理吗？";
		next;
		.@totalcost = .@repairprice*.@checkitem;
		mes "["+ .@npc_name$ +"]";
		mes "每件修理费是 " + .@repairprice + " Zeny。";
		mes "修理全部损坏的物品需要 " + .@totalcost + " Zeny！要修理吗？";
		next;
		switch(select("是:否")) {
		case 1:
			if (Zeny < .@totalcost) {
				mes "["+ .@npc_name$ +"]";
				mes "哎呀呀……";
				mes "付修理费之前先看看钱包吧！";
				mes "你的 Zeny 不够，我什么都修不了。";
				close;
			}
			.@checkitem2 = 1;
			while (1) {
				if (getbrokenid(.@checkitem2) == 0) {
					break;
				}
				.@checkitem2 = .@checkitem2+1;
			}
			.@checkitem2 = .@checkitem2-1;
			if (.@checkitem == .@checkitem2) {
				Zeny = Zeny-.@totalcost;
				while (.@checkitem) {
					repair(.@checkitem);
					.@checkitem = .@checkitem-1;
				}
				mes "["+ .@npc_name$ +"]";
				mes "好了！全部修好了。";
				mes "以后小心一点吧。";
				mes "物品也是有生命的，你知道吧。";
				close;
			}
			else {
				mes "["+ .@npc_name$ +"]";
				mes "嗯？有点不对劲。";
				mes "等等……把要修理的物品装备上，再来找我。";
				close;
			}
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "好吧，虽然不关我的事，";
			mes "但物品损坏了放着可不好。";
			mes "应该尽快修理才对！";
			close;
		}
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "呵呵呵……";
		mes "没有要修理的物品，";
		mes "就没什么好找我";
		mes "的了。";
		close;
	}
}
