prontera,73,140,0	script	牛奶商人	86,{
	if (checkweight(1201,1) == 0) {
		mes "^3355FF请稍等一下！";
		mes "您身上的物品实在太多了，";
		mes "我没办法为您服务。";
		mes "请把多余的物品存进卡普拉仓库，";
		mes "然后再来吧~";
		close;
	}
	mes "[牛奶商人]";
	mes "嘿，嘿……";
	mes "只要带 1 个空瓶和 15 Zeny 给我，";
	mes "我就换给您 1 瓶牛奶。";
	mes "怎么样，划算吧？";
	next;
	if (select("把空瓶全部换掉。:取消") == 1) {
		if (countitem(713) <= 0) {
			mes "[牛奶商人]";
			mes "嘿……您身上一个空瓶都没有。";
			mes "没有空瓶的话，我也没办法把牛奶换给您啊……";
			close;
		}
		set .@bottles,countitem(713);
		set .@total_weight,.@bottles * 50;
		set .@total_cost,.@bottles * 15;
		if (Zeny < .@Total_cost) {
			mes "[牛奶商人]";
			mes "哦，哎呀~！";
			mes "您的 Zeny 不够把这些空瓶全部换成牛奶。";
			mes "至少需要准备 "+ .@total_cost +" Zeny 才行。";
			close;
		}
		if (MaxWeight - Weight < .@total_weight) {
			mes "[牛奶商人]";
			mes "嗯……";
			mes "在我把这些牛奶全部交给您之前，";
			mes "能不能先在物品栏里腾出一点空间呢？";
			close;
		}
		set Zeny, Zeny-.@total_cost;
		delitem 713,.@bottles; //Empty Bottles
		getitem 519,.@bottles; //Milk
		close;
	}
	close;
}
