lhz_in02,230,284,4	script	饭店 职员#01	86,{
	mes "[饭店 职员]";
	mes "欢迎光临皇家金龙饭店。";
	mes "这里有最舒适的住宿环境和最优质的服务。";
	next;
	switch(select("储存重生点:休息 - 5,000 Zeny:取消")) {
	case 1:
		savepoint "lhz_in02",209,275,1,1;
		mes "[饭店 职员]";
		mes "谢谢，您的重生点已经储存在皇家金龙饭店了。";
		close;
	case 2:
		if(Zeny < 5000) {
			mes "[饭店 职员]";
			mes "非常抱歉，入住需要 5,000 Zeny。";
			close;
		}
		set Zeny,Zeny - 5000;
		mes "[饭店 职员]";
		mes "谢谢~ 祝您在皇家金龙饭店住得愉快。";
		close2;
		warp "lhz_in02",219,150;
		percentheal 100,100;
		npcskill "AL_BLESSING",10,99,99;
		end;
	case 3:
		mes "[饭店 职员]";
		mes "谢谢，祝您愉快。";
		close;
	}
}

//======================= Inn Function ==============================
function	script	F_InnMaid	{
	.@npc_name$ = getarg(0);

	mes .@npc_name$;
	mes "欢迎光临";
	mes getarg(1) + "。";
	mes "请问有什么可以帮您？";
	next;
	switch(select("储存重生点:休息 -> 5000 Zeny:取消")) {
	case 1:
		mes .@npc_name$;
		mes "您的重生点已经储存好了。";
		mes "谢谢，请再来。";
		savepoint getarg(2),getarg(3),getarg(4),1,1;
		close;
	case 2:
		mes .@npc_name$;
		if(Zeny < 5000){
			mes "非常抱歉，服务费是 5,000 Zeny。";
			mes "下次请带够足够的钱再来入住，好吗？";
			close;
		}
		mes "谢谢，祝您休息愉快~";
		close2;
		set Zeny,Zeny - 5000;
		percentheal 100,100;
		return;
	case 3:
		close;
	}
}
