payon_in01,5,134,5	script	发明家 钦斯	89,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
		mes "[发明家 钦斯]";
		mes "喂，您身上的东西也太多了吧。";
		mes "先存进卡普拉仓库如何？";
		mes "弄好了我们再聊，";
		mes "好吧？";
		close;
	}
	mes "[发明家 钦斯]";
	mes "我叫钦斯。不是我自夸，";
	mes "我大概就是 ^663300当代最伟大的发明家^000000。";
	next;
	mes "[发明家 钦斯]";
	mes "这一次，我造出了真正不得了的东西。";
	mes "我把它叫做…… ^663300魔法箭筒^000000！！";
		mes "这必将被载入史册，";
	mes "一场箭矢的革命！";
	emotion ET_BEST;
	next;
	mes "[发明家 钦斯]";
	mes "我研究魔法和箭筒多年，夜以继日，";
	mes "终于想出了用魔法压缩箭矢的办法！";
	mes "有了魔法箭筒，您能带更多的箭，";
	mes "重量却更轻！";
	next;
	mes "[发明家 钦斯]";
	mes "要不要试试我的箭筒？我相信，";
	mes "像您这样有眼光的人，一定懂得我的天才！";
	next;
	switch(select("箭矢筒:铁箭矢筒:钢铁箭矢筒:神之金属箭矢筒:火箭矢筒:银箭矢筒:风灵箭矢筒:地灵箭矢筒:水灵箭矢筒:影子箭矢筒:无形箭矢筒:铁锈箭矢筒")) {
	case 1: callsub S_BuyQuiver,1750,500,500,12004;
	case 2: callsub S_BuyQuiver,1770,500,500,12005;
	case 3: callsub S_BuyQuiver,1753,500,500,12006;
	case 4: callsub S_BuyQuiver,1765,500,500,12007;
	case 5: callsub S_BuyQuiver,1752,500,500,12008;
	case 6: callsub S_BuyQuiver,1751,500,500,12009;
	case 7: callsub S_BuyQuiver,1755,500,500,12010;
	case 8: callsub S_BuyQuiver,1756,500,500,12011;
	case 9: callsub S_BuyQuiver,1754,500,500,12012;
	case 10: callsub S_BuyQuiver,1767,500,500,12013;
	case 11: callsub S_BuyQuiver,1757,500,500,12014;
	case 12: callsub S_BuyQuiver,1762,500,500,12015;
	}
	//This doesn't seem to be accessable...
	//mes "[Inventor Jaax]";
	//mes "Is there";
	//mes "anything you want?";
	//close;

S_BuyQuiver:
	.@item_id = getarg(0);
	.@amount = getarg(1);
	.@zeny_req = getarg(2);

	if (countitem(.@item_id) >= .@amount) {
		mes "[发明家 钦斯]";
		mes "太好了！";
		mes "您身上带着箭矢吧？只要 ^FF3131"+ .@zeny_req +" Zeny^000000，我就给您一个能装 "+ .@amount +" 支 "+ getitemname(.@item_id) +" 的箭筒。";
		next;
		switch(select("把箭全部装进箭筒:买 1 个箭筒:取消")) {
		case 1:
			.@arrows = countitem(.@item_id);
			set .@quiver,.@arrows / .@amount;
			set .@arrows_used,.@quiver * .@amount;
			.@arrow_zeny01 = .@quiver * .@zeny_req;
			mes "箭矢数量：^3131FF"+.@arrows+" ^000000";
			mes "最多可购买箭筒数量：^3131FF"+.@quiver+" ^000000";
			mes "所需 Zeny：^3131FF"+.@arrow_zeny01+" Zeny^000000";
			next;
			mes "[发明家 钦斯]";
			mes "要把您现在身上的箭，能装多少就装多少，";
			mes "全部换成箭筒吗？";
			next;
			if (select("是:取消") == 1) {
				if (.@arrow_zeny01 < Zeny) {
					mes "[发明家 钦斯]";
					mes "拿去吧！";
					mes "记住，^FF0000当负重达到上限的 90% 时，";
					mes "就无法使用箭筒了^000000。";
					set Zeny, Zeny-.@arrow_zeny01;
					delitem .@item_id,.@arrows_used;// Arrow
					getitem getarg(3),.@quiver; //Quiver
					next;
					mes "[发明家 钦斯]";
					mes "那么……平时多留意一下自己的负重，";
					mes "就不会有什么问题。";
					close;
				}
				else {
					mes "[发明家 钦斯]";
					mes "抱歉，您的 Zeny 不够。";
					mes "这可是我花了好多年才开发出来的革命性技术，";
					mes "总不能白送吧！";
					close;
				}
			}
			mes "[发明家 钦斯]";
			mes "您改变主意了？";
			mes "明明拥有一个箭筒的荣耀已经近在咫尺了？";
			close;
		case 2:
			if (Zeny > .@zeny_req) {
				mes "[发明家 钦斯]";
				mes "拿去吧！";
				mes "记住，^FF0000当负重达到上限的 90% 时，";
				mes "就无法使用箭筒了^000000。";
				Zeny = Zeny -.@zeny_req;
				delitem .@item_id,.@amount;// Arrow
				getitem getarg(3),1; //Quiver
				next;
				mes "[发明家 钦斯]";
				mes "那么……平时多留意一下自己的负重，";
				mes "就不会有什么问题。";
				close;
			}
			else {
				mes "[发明家 钦斯]";
				mes "您竟然连 "+ .@zeny_req +" Zeny 都没有？";
				mes "实在太遗憾了。";
				mes "我没想到您居然这么……";
				mes "^333333穷困潦倒^000000。";
				close;
			}
		case 3:
			mes "[发明家 钦斯]";
			mes "什么……？";
			mes "您难道看不出这项发明";
		mes "将永远改变携带箭矢的方式吗？！";
			mes "未来就在眼前啊！";
			close;
		}
	}
	else {
		mes "[发明家 钦斯]";
		mes "这个箭筒最多能装 "+ .@amount +" 支箭。它用的是我的独门秘方，";
		mes "箭加箭筒的总重量，比单带箭还要轻。";
		next;
		mes "[发明家 钦斯]";
		mes "这可是科学的奇迹！只要带至少 "+ .@amount +" 支箭和每个箭筒 "+ .@zeny_req +" Zeny 来，您就能亲身体验。";
		close;
	}
}
