alde_alche,24,188,3	script	公会销售员	740,{
	if (checkweight(1201,1) == 0) {
		mes "- 请稍等一下！ -";
		mes "- 现在您身上的物品实在太多了。 -";
		mes "- 请把一部分物品存进卡普拉仓库之后再来吧。";
		mes " -";
		close;
	}
	mes "[盖弗 艾尔 夏普]";
	mes "欢迎来到炼金术士公会。";
	mes "今天需要点什么？";
	next;
	switch(select("购买材料。:购买制作说明书。:离开。")) {
	case 1:
		mes "[盖弗 艾尔 夏普]";
		mes "您想要什么？";
		next;
		if (select("药钵 - 8 Zeny:取消。") == 1) {
			mes "[盖弗 艾尔 夏普]";
			mes "您要几个？不想买的话请输入 '0'。";
			next;
			while(1) {
				input .@input,0,2001;
				if (.@input == 0) {
					mes "[盖弗 艾尔 夏普]";
					mes "交易取消了。下次请再来。";
					close;
				}
				else if (.@input < 1 || .@input > 2000) {
					mes "[盖弗 艾尔 夏普]";
					mes "数量必须在 2000 以内。";
					next;
				}
				else {
					break;
				}
			}
			set .@sell,.@input * 8;
			set .@item_weight,.@input * 10;
			if (Zeny < .@sell) {
				mes "[盖弗 艾尔 夏普]";
				mes "您的 Zeny 不够。";
				mes "请先确认一下自己有多少 Zeny。";
				close;
			}
			if ((MaxWeight - Weight) < .@item_weight) {
				mes "[盖弗 艾尔 夏普]";
				mes "看来这些东西您没办法全部带走。";
				mes "请确认一下物品栏的空间。";
				close;
			}
			set Zeny, Zeny-.@sell;
			getitem 7134,.@input; //Medicine_Bowl
			mes "[盖弗 艾尔 夏普]";
			mes "谢谢。欢迎再来。";
			close;
		}
		mes "[盖弗 艾尔 夏普]";
		mes "那么，需要购买炼金相关材料的时候，";
		mes "请再来吧，好吗？";
		close;
	case 2:
		mes "[盖弗 艾尔 夏普]";
		mes "您需要哪一本？";
		mes "制作说明书一般是 100,000 Zeny。";
		mes "不过有几本比较特殊，";
		mes "价格会贵一些。";
		next;
		switch(select("一般药水制作说明书:酒精制作说明书:火烟瓶制作说明书:盐酸瓶制作说明书:植物瓶制作说明书:刺锤瓶制作说明书:护贝药制作说明书:纤细药水制作说明书:取消交易。")) {
		case 1:
			callsub S_SellManual,7144,100000; //Normal_Potion_Book
			break;
		case 2:
			callsub S_SellManual,7127,100000; //Alcol_Create_Book
			break;
		case 3:
			callsub S_SellManual,7128,100000; //FireBottle_Create_Book
			break;
		case 4:
			callsub S_SellManual,7129,100000; //Acid_Create_Book
			break;
		case 5:
			callsub S_SellManual,7130,100000; //Plant_Create_Book
			break;
		case 6:
			callsub S_SellManual,7131,100000; //Mine_Create_Book
			break;
		case 7:
			callsub S_SellManual,7132,100000; //Coating_Create_Book
			break;
		case 8:
			callsub S_SellManual,7133,240000; //Slim_Potion_Create_Book
			break;
		case 9:
			mes "[盖弗 艾尔 夏普]";
			mes "那么……需要购买制作说明书的时候，";
			mes "请再来吧。";
			close;
		}
	case 3:
		mes "[盖弗 艾尔 夏普]";
		mes "好的，祝您愉快。";
		close;
	}

S_SellManual:
	.@item_id = getarg(0);
	.@zeny_req = getarg(1);

	if (Zeny < .@zeny_req) {
		mes "[盖弗 艾尔 夏普]";
		mes "您的 Zeny 不够。";
		mes "请先确认一下自己有多少 Zeny。";
		close;
	}
	mes "[盖弗 艾尔 夏普]";
	mes getitemname(.@item_id) +"吗？";
	mes "价格是 "+ .@zeny_req +" Zeny。";
	next;
	if (select("购买。:离开。") == 1) {
		Zeny = Zeny - .@zeny_req;
		getitem .@item_id,1;
		mes "[盖弗 艾尔 夏普]";
		mes "谢谢惠顾。";
		close;
	}
	mes "[盖弗 艾尔 夏普]";
	mes "下次请再来。";
	close;
}
