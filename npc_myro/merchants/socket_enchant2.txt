		mes "^FF0000当然，还需要一件 "+ getitemname(.@item_id) +"。^000000";
	}
	else
	{
		mes getitemname(.@item_id) +"？好的，那你需要带来";
		mes "^FF0000"+ (.@zeny_req >= 1000 ? (.@zeny_req/1000 +",000") : .@zeny_req) +",000 Zeny 作为手续费，以及 "+ .@req_amount[0] +" 个"+ getitemname(.@item_req[0]) +"作为材料。";

					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					getitem getarg(1),1;
					next;
					mes "[雷亚布连]";

				{

					mes "哇！……非常抱歉，失败了。";

					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
