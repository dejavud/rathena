	function Dyes {
	setarray .@DyeName$[0],"原色","赤色","紫色","红色","黑色","绿色","蓝色","白色","黄色";
	setarray .@DyeItems[0],0,980,981,975,983,979,978,982,976;
	set .@menu$,"- 原色:";
	for(set .@i,0; .@i<getargcount(); set .@i,.@i+2)
		set .@menu$, .@menu$+"- "+.@DyeName$[getarg(.@i)]+":";
	set .@menu$, .@menu$+"- ^777777取消^000000";
	set .@s, select(.@menu$)-1;
	if (.@s == (getargcount()/2)+1) goto L_End;
	mes .n$;
	if (.@s == 0) {
		mes "我可以免费把您的衣服";
		mes "染回原色。";
		mes "确定吗？";
		next;
		switch(select("确定:取消")) {
		case 1:
			mes .n$;
			setlook 7,0;
			mes "您的衣服已经染成"+.@DyeName$[0]+"了。";
			close;
		case 2:
			goto L_End;
		}
	}
	if (!countitem(.@DyeItems[getarg((.@s-1)*2)])) {
		mes "要给您的衣服染色，";
		mes "我需要相应的染料。";
		mes "详情请看我的价目表。";
		close;
	}
	if (Zeny < 10000) {
		mes "抱歉，您的钱不够。";
		close;
	}
	delitem .@DyeItems[getarg((.@s-1)*2)],1;
	set Zeny, Zeny-10000;
	setlook 7, ((.@s)?(getarg((.@s-1)*2+1)):0);
	mes "您的衣服已经染成"+((.@s)?.@DyeName$[getarg((.@s-1)*2)]:.@DyeName$[0])+"了。";
	close;
}
