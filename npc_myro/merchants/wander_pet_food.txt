	mes "[宠物商人 珍妮]";

	mes "真的很难。";
	next;
	mes "[宠物商人 珍妮]";
	mes "你有没有正在找的食物呀？";
	next;
	switch(select("有:没有")) {
	case 1:

			mes "[宠物商人 珍妮]";
			mes "哦，";

			next;
			switch(select("好的，拜托了！:不用了，谢谢。")) {
			case 1:
				mes "[宠物商人 珍妮]";
				mes "好，拿去吧。";
				mes "希望它能让你可爱的宠物开开心心。";
				delitem 7158,20; //Broken_Wine_Vessel

				mes "[宠物商人 珍妮]";
				mes "呵呵，改变主意的话随时跟我说哦。";
				close;
			}
		}

			mes "[宠物商人 珍妮]";
			mes "其实，现在我可以帮你弄到";
			mes "邪骸浪人宠物的食物——灵魂酒。";

			mes "我就给你 ^FF0000 20 份灵魂酒^000000。";
			next;
			mes "破碎的酒缸可以从天津洞穴里的天狗身上打到。";

		}
	case 2:
		close;
	}
