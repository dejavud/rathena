lhz_airport,143,43,5	script	机场职员#lhz_air1a::lhz_airport1	90,{
	mes "[机场职员]";
	mes "欢迎来到里希塔乐镇机场，";
	mes "我们提供直飞艾音布罗克、";
	mes "朱诺和毁葛的航线。";
	next;
	if (select("搭乘飞空艇。:取消。") == 1) {
		mes "[机场职员]";
		mes "登船费为 1,200 zeny，";
		mes "使用飞空艇免费搭乘券可以免费。";
		next;
		if (select("是:否") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "lhz_airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "lhz_airport",148,51;
				end;
			}
			mes "[飞空艇乘务员]";
			mes "抱歉，您没有足够的";
			mes "1,200 zeny 来支付登船费。";
			close;
		}
	}
	mes "[机场职员]";
	mes "谢谢，欢迎下次再来。";
	mes "祝您愉快~";
	close;
}

lhz_airport,158,43,5	duplicate(lhz_airport1)	机场职员#lhz_air1b	90
lhz_airport,126,43,5	duplicate(lhz_airport1)	机场职员#lhz_air1c	90

lhz_airport,143,49,3	script	到达职员#lhz_air2a::lhz_airport2	90,{
	mes "[到达职员]";
	mes "欢迎来到里希塔乐镇机场。";
	mes "如果您是刚下飞空艇的旅客，我可以带您前往主航站楼。";
	mes "否则，请搭乘飞空艇前往您的目的地。";
	next;
	if (select("前往主航站楼。:取消。") == 1) {
		mes "[到达职员]";
		mes "一旦进入主航站楼，再次搭乘飞空艇时需要重新付费。";
		mes "只有当里希塔乐镇是您的目的地时才应该出去。";
		mes "确定前往主航站楼吗？";
		next;
		if (select("是:否") == 1) {
			warp "lhz_airport",142,40;
			end;
		}
	}
	mes "[到达职员]";
	mes "好的，谢谢惠顾，";
	mes "祝您旅途愉快~";
	close;
}
