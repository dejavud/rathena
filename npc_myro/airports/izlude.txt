-	script	::Airship_Staff_izlude	-1,{
	mes "[飞空艇乘务员]";
	mes "欢迎！您要搭乘";
	mes "途经朱诺和拉赫的";
	mes "飞空艇吗？";
	next;
	if (select("搭乘飞往朱诺/拉赫的飞空艇。:取消。") == 1) {
		mes "[飞空艇乘务员]";
		mes "登船费为 1,200 zeny，";
		mes "使用飞空艇免费搭乘券可以免费。";
		mes "请问您要登船吗？";
		next;
		if (select("是:否") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "airplane_01",244,58;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "airplane_01",244,58;
				end;
			}
			mes "[飞空艇乘务员]";
			mes "抱歉，您没有足够的";
			mes "1,200 zeny 来支付登船费。";
			close;
		}
	}
	mes "[飞空艇乘务员]";
	mes "谢谢，欢迎下次再来。";
	mes "祝您愉快~";
	close;
}
