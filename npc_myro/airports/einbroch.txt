airport,143,43,5	script	机场职员#airport1a::airport1	90,{
	mes "[机场职员]";
	mes "欢迎来到艾音布罗克机场，";
	mes "我们提供直飞朱诺、";
	mes "里希塔乐镇和毁葛的航线。";
	next;
	if (select("搭乘飞空艇:取消") == 1) {
		mes "[机场职员]";
		mes "登船费为 1,200 zeny，";
		mes "使用飞空艇免费搭乘券可以免费。";
		mes "您要登船吗？";
		next;
		if (select("是:否") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "airport",148,51;
				end;
			}
			mes "[机场职员]";
			mes "抱歉，您没有飞空艇免费搭乘券，";
			mes "也没有足够的 zeny 支付登船费。";
			mes "请记住，登船费是 1,200 zeny。";
			close;
		}
	}
	mes "[机场职员]";
	mes "谢谢，祝您愉快。";
	close;
}
