y_airport,143,43,5	script	机场职员#y_air1a::y_airport1	90,{
	mes "[机场职员]";
	mes "欢迎来到朱诺机场。";
	mes "我们提供前往艾音布罗克、里希塔乐镇和毁葛的国内航线，";
	mes "以及前往依斯鲁得岛和拉赫的国际航线。";
	mes "请问有什么可以帮您？";
	next;
	if (select("搭乘飞空艇。:取消。") == 1) {
		mes "[机场职员]";
		mes "所有航班的登船费都是 1,200 zeny。";
		mes "使用飞空艇免费搭乘券可以免费。";
		mes "请问您要出发吗？";
		next;
		if (select("是:否") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "y_airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "y_airport",148,51;
				end;
			}
			mes "[机场职员]";
			mes "抱歉，您没有足够的";
			mes "1,200 zeny 来支付登船费。";
			close;
		}
	}
	mes "[机场职员]";
	mes "谢谢，祝您愉快。";
	close;
}

	mes "否则，请搭乘飞空艇前往您的目的地。";
	next;
	if (select("前往主航站楼:取消") == 1) {
		mes "[机场职员]";
		mes "一旦进入主航站楼，再次搭乘飞空艇时需要重新付费。";
		mes "只有当朱诺是您的目的地时才应该出去。";
		mes "确定要前往主航站楼吗？";
		next;
		if (select("是:否") == 1) {
			warp "y_airport",142,40;
			end;
		}
	}
	mes "[机场职员]";
	mes "好的，谢谢惠顾，";
	mes "祝您旅途愉快~";
	close;
}
