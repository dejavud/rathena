valkyrie,53,58,3	script	神行太保#Valkyrie	747,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[神行太保]";
			mes "恭喜你。";
			mes "荣耀归于战士们！";
			close;
		}
		mes "[神行太保]";
		mes "嘿……";
		mes "想当英雄，又改不掉";
		mes "阴险、狡诈、偷偷摸摸";
		mes "的本性……可真难啊……";
		next;
		mes "[神行太保]";
		mes "但每到艰难时刻，";
		mes "伙伴们都知道可以指望我。";
		mes "我需要他们，";
		mes "他们也需要我。";
		close;
	}
	else if (ADVJOB == Job_Stalker && Class == Job_Thief_High && JobLevel > 39) {
		mes "[神行太保]";
		mes "这个世界需要更多";
		mes "愿意游走于秩序与";
		mes "混乱边缘的英雄。";
		next;
		mes "[神行太保]";
		mes "你准备好加入";
		mes "最狡黠的战士行列了吗？";
		mes "你准备好成为神行太保了吗？";
		next;
		if (select("不。:是。") == 1) {
			mes "[神行太保]";
			mes "等你准备好了，";
			mes "随时回来找我。";
			mes "荣耀归于战士们！";
			close;
		}
		if (SkillPoint) {
			mes "[神行太保]";
			mes "你还有技能可以学习。";
			mes "请先用完所有剩余的技能点，再来找我。";
			close;
		}
		jobchange Job_Stalker;
		set ADVJOB,0;
		mes "[神行太保]";
		mes "恭喜！";
		mes "作为神行太保，我希望你";
		mes "从背后捅对人。";
		mes "用恶徒自己的卑鄙手段";
		mes "去消灭他们吧！";
		close;
	}
	mes "[神行太保]";
	mes "欢迎来到瓦尔哈拉，";
	mes "这里是荣誉殿堂。";
	next;
	mes "[神行太保]";
	mes "待在这里的期间，";
	mes "请随意一些。荣耀归于战士们！";
	close;
}
