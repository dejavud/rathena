valkyrie,53,42,3	script	神拳大师#Valkyrie	52,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[神拳大师]";
			mes "恭喜你。";
			mes "荣耀归于战士们！";
			close;
		}
		mes "[神拳大师]";
		mes "技巧、速度、力量、敏捷。";
		next;
		mes "[神拳大师]";
		mes "神拳大师能兼得";
		mes "这些能力带来的益处。";
		mes "但人这一生能精通的，";
		mes "实在有限……";
		close;
	}
	else if (ADVJOB == Job_Champion && Class == Job_Acolyte_High && JobLevel > 39) {
		mes "[神拳大师]";
		mes "时机已到。";
		mes "伟大的英雄们该挺身而出，";
		mes "对抗肆虐米德加兹的";
		mes "邪恶势力了！";
		next;
		mes "[神拳大师]";
		mes "你想成为神拳大师吗？";
		next;
		if (select("不。:是。") == 1) {
			mes "[神拳大师]";
			mes "等你准备好了，";
			mes "随时回来找我。";
			mes "荣耀归于战士们！";
			close;
		}
		if (SkillPoint) {
			mes "[神拳大师]";
			mes "你还有技能可以学习。";
			mes "请先用完所有剩余的技能点，再来找我。";
			close;
		}
		jobchange Job_Champion;
		set ADVJOB,0;
		mes "[神拳大师]";
		mes "恭喜！";
		mes "以神拳大师的身份活下去吧，";
		mes "用你双拳的力量，";
		mes "为世界带来光明。";
		close;
	}
	mes "[神拳大师]";
	mes "欢迎来到瓦尔哈拉，";
	mes "这里是荣誉殿堂。";
	next;
	mes "[神拳大师]";
	mes "待在这里的期间，";
	mes "请随意一些。荣耀归于战士们！";
	close;
}
