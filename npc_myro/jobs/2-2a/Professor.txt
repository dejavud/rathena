valkyrie,53,47,3	script	教授#Valkyrie	743,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[教授]";
			mes "恭喜你。";
			mes "荣耀归于战士们！";
			close;
		}
		mes "[教授]";
		mes "需要花上一辈子……";
		mes "真的是一辈子，";
		mes "才能积累成为教授";
		mes "所需的知识……";
		next;
		mes "[教授]";
		mes "学无止境，令人敬畏。";
		mes "你学得越多，";
		mes "就越发现自己无知的地方。";
		mes "学习的过程永无止境……";
		close;
	}
	else if (ADVJOB == Job_Professor && Class == Job_Mage_High && JobLevel > 39) {
		mes "[教授]";
		mes "米德加兹没有足够的教授";
		mes "来引领繁荣的新时代。";
		mes "这个世界需要你……";
		next;
		mes "[教授]";
		mes "你愿意承担这份";
		mes "重大的责任吗？";
		mes "你愿意以教授的身份";
		mes "为米德加兹服务吗？";
		next;
		if (select("不。:是。") == 1) {
			mes "[教授]";
			mes "等你准备好了，";
			mes "随时回来找我。";
			mes "荣耀归于战士们！";
			close;
		}
		if (SkillPoint) {
			mes "[教授]";
			mes "你还有技能可以学习。";
			mes "请先用完所有剩余的技能点，再来找我。";
			close;
		}
		jobchange Job_Professor;
		set ADVJOB,0;
		mes "[教授]";
		mes "恭喜！";
		mes "作为教授，我希望你";
		mes "积极地将知识之光";
		mes "带到黑暗之处。";
		close;
	}
	mes "[教授]";
	mes "欢迎来到瓦尔哈拉，";
	mes "这里是荣誉殿堂。";
	next;
	mes "[教授]";
	mes "待在这里的期间，";
	mes "请随意一些。荣耀归于战士们！";
	close;
}
