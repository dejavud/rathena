valkyrie,53,54,3	script	乐师#Valkyrie	741,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[乐师]";
			mes "恭喜你。";
			mes "荣耀归于战士们！";
			close;
		}
		mes "[乐师]";
		mes "想和我一起唱首歌吗？";
		mes "沙啦啦啦啦～";
		close;
	}
	else if (ADVJOB == Job_Clown && Class == Job_Archer_High && JobLevel > 39) {
		mes "[乐师]";
		mes "沉闷的凡间";
		mes "需要更多欢快的歌声。";
		mes "你愿意把它带给世人，";
		mes "扭转与邪恶之战的局势吗？";
		next;
		mes "[乐师]";
		mes "你愿意为了米德加兹……";
		mes "以乐师的身份这么做吗？";
		next;
		if (select("不。:是。") == 1) {
			mes "[乐师]";
			mes "等你准备好了，";
			mes "随时回来找我。";
			mes "荣耀归于战士们！";
			close;
		}
		if (SkillPoint) {
			mes "[乐师]";
			mes "你还有技能可以学习。";
			mes "请先用完所有剩余的技能点，再来找我。";
			close;
		}
		jobchange Job_Clown;
		set ADVJOB,0;
		mes "[乐师]";
		mes "恭喜！";
		mes "作为乐师，你的歌声";
		mes "将为同伴带来希望，";
		mes "为敌人带来绝望。";
		close;
	}
	mes "[乐师]";
	mes "欢迎来到瓦尔哈拉，";
	mes "这里是荣誉殿堂。";
	next;
	mes "[乐师]";
	mes "待在这里的期间，";
	mes "请随意一些。荣耀归于战士们！";
	close;
}
