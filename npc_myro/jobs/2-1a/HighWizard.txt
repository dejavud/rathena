valkyrie,44,47,5	script	超魔导师#Valkyrie	735,{
	if (ADVJOB == 0 ||  Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[超魔导师]";
			mes "恭喜。";
			mes "荣耀归于战士们！";
			close;
		}
		mes "[超魔导师]";
		mes "我们超魔导师肩负着责任，";
		mes "要将毁灭性的魔法用于正途。";
		next;
		mes "[超魔导师]";
		mes "成为超魔导师需要毕生的修行。";
		mes "你能想象如果我们的力量";
		mes "落入坏人之手会怎样吗？！";
		close;
	}
	else if (ADVJOB == Job_High_Wizard && Class == Job_Mage_High && JobLevel > 39) {
		mes "[超魔导师]";
		mes "时机已到。";
		mes "米德加兹需要能够";
		mes "驾驭最强魔法的人……";
		next;
		mes "[超魔导师]";
		mes "你想成为超魔导师吗？";
		next;
		if (select("不。:是的。") == 1) {
			mes "[超魔导师]";
			mes "等你准备好了，";
			mes "随时回来吧。";
			mes "荣耀归于战士们！";
			close;
		}
		if (SkillPoint) {
			mes "[超魔导师]";
			mes "你还可以学习更多技能。";
			mes "请把剩余的技能点全部用完，再来找我。";
			close;
		}
		jobchange Job_high_Wizard;
		set ADVJOB,0;
		mes "[超魔导师]";
		mes "恭喜。";
		mes "作为超魔导师，";
		mes "愿你用力量";
		mes "为受压迫者带来安宁。";
		close;
	}
	mes "[超魔导师]";
	mes "欢迎来到瓦尔哈拉，";
	mes "荣誉殿堂。";
	next;
	mes "[超魔导师]";
	mes "在这里请随意。";
	mes "荣耀归于战士们！";
	close;
}
