valkyrie,44,39,5	script	骑士领主#Valkyrie	56,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[骑士领主]";
			mes "恭喜。";
			mes "荣耀归于战士们！";
			close;
		}
		mes "[骑士领主]";
		mes "我们骑士肩负着";
		mes "重大的责任……";
		mes "服务与守护。";
		next;
		mes "[骑士领主]";
		mes "即使付出生命，";
		mes "我们也要守护战友的安危。";
		close;
	}
	else if (ADVJOB == Job_Lord_Knight && Class == Job_Swordman_High && JobLevel > 39) {
		mes "[骑士领主]";
		mes "你的时机已到！";
		mes "世界仍需要你。";
		mes "请以全新的面貌，";
		mes "继续你的英雄之路。";
		next;
		mes "[骑士领主]";
		mes "你想成为骑士领主吗？";
		next;
		if (select("不。:是的。") == 1) {
			mes "[骑士领主]";
			mes "等你准备好了，";
			mes "随时回来吧。";
			mes "荣耀归于战士们！";
			close;
		}
		if (SkillPoint) {
			mes "[骑士领主]";
			mes "你还可以学习更多技能。";
			mes "请把剩余的技能点全部用完，再来找我。";
			close;
		}
		jobchange Job_Lord_Knight;
		set ADVJOB,0;
		mes "[骑士领主]";
		mes "恭喜！";
		mes "作为骑士领主，";
		mes "愿你身先士卒，";
		mes "引领盟友走向胜利！";
		close;
	}
	mes "[骑士领主]";
	mes "欢迎来到瓦尔哈拉，";
	mes "荣誉殿堂。";
	next;
	mes "[骑士领主]";
	mes "在这里请随意。";
	mes "荣耀归于战士们！";
	close;
}
