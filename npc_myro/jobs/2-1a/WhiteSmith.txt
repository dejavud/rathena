valkyrie,44,50,5	script	银匠#Valkyrie	731,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[银匠]";
			mes "恭喜。";
			mes "荣耀归于战士们！";
			close;
		}
		mes "[银匠]";
		mes "啊……";
		mes "工艺的巅峰。";
		mes "这就是银匠的杰作。";
		next;
		mes "[银匠]";
		mes "当你成为精英中的精英，";
		mes "便开始为了探索更新、";
		mes "更好的工艺而锻造……";
		close;
	}
	else if (ADVJOB == Job_WhiteSmith && Class == Job_Merchant_High && JobLevel > 39) {
		mes "[银匠]";
		mes "时机已到！";
		mes "我们的世界需要你这样";
		mes "勇敢、勤劳的冒险者……";
		next;
		mes "[银匠]";
		mes "你想成为银匠吗？";
		next;
		if (select("不。:是的。") == 1) {
			mes "[银匠]";
			mes "等你准备好了，";
			mes "随时回来吧。";
			mes "荣耀归于战士们！";
			close;
		}
		if (SkillPoint) {
			mes "[银匠]";
			mes "你还可以学习更多技能。";
			mes "请把剩余的技能点全部用完，再来找我。";
			close;
		}
		jobchange Job_Whitesmith;
		set ADVJOB,0;
		mes "[银匠]";
		mes "恭喜！";
		mes "作为银匠，";
		mes "愿你为米德加兹锻造出";
		mes "通往更光明未来的道路。";
		close;
	}
	mes "[银匠]";
	mes "欢迎来到瓦尔哈拉，";
	mes "荣誉殿堂。";
	next;
	mes "[银匠]";
	mes "在这里请随意。";
	mes "荣耀归于战士们！";
	close;
}
