valkyrie,44,55,5	script	神射手#Valkyrie	727,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[神射手]";
			mes "恭喜。";
			mes "荣耀归于战士们！";
			close;
		}
		mes "[神射手]";
		mes "一击必杀。";
		mes "一旦你领悟了这种战斗方式，";
		mes "就并不困难。";
		close;
	}
	else if (ADVJOB == Job_Sniper && Class == Job_Archer_High && JobLevel > 39) {
		mes "[神射手]";
		mes "世界需要你这样";
		mes "强大的弓箭手。";
		mes "你准备好肩负这份重任了吗？";
		next;
		mes "[神射手]";
		mes "你愿意更进一步，";
		mes "成为神射手吗？";
		next;
		if (select("不。:是的。") == 1) {
			mes "[神射手]";
			mes "等你准备好了，";
			mes "随时回来吧。";
			mes "荣耀归于战士们！";
			close;
		}
		if (SkillPoint) {
			mes "[神射手]";
			mes "你还可以学习更多技能。";
			mes "请把剩余的技能点全部用完，再来找我。";
			close;
		}
		jobchange Job_Sniper;
		set ADVJOB,0;
		mes "[神射手]";
		mes "恭喜！";
		mes "作为神射手，愿邪恶爪牙";
		mes "只要落入你的视野，";
		mes "就再无安宁之日！";
		close;
	}
	mes "[神射手]";
	mes "欢迎来到瓦尔哈拉，";
	mes "荣誉殿堂。";
	next;
	mes "[神射手]";
	mes "在这里请随意。";
	mes "荣耀归于战士们！";
	close;
}
