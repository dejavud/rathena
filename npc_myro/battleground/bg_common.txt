		mes "[艾伦德克]";
		if (.@j == 3) {
			mes "你取消了兑换。";
			break;

		set .@cost, ((.@j==1)?7828:7829);
		if (countitem(.@cost) >= .@Items[.@i+1]) {
			mes "感谢兑换。";
			delitem .@cost, .@Items[.@i+1];
			getitem .@Items[.@i],1;
		}
		else mes "你的"+getitemname(.@cost)+"不够。";
		break;
	}
	close;
}

// Badge Reward Function
//============================================================
function	script	F_BG_Badge	{
	.@team$ = getarg(1);
	.@bg_type$ = getarg(2);

	if (.@bg_type$ == "Tierra") {
		set .@badge,7828; //BF_Badge1
		set .@amount_win,3;
		set .@amount_lose,1;
	} else if (.@bg_type$ == "Flavius") {
		set .@badge,7829; //BF_Badge2
		set .@amount_win,9;
		set .@amount_lose,3;
	} else
		return;

	if (getarg(0) == 1) {
		set .@amount, .@amount_win;
		if (.@team$ == "Guillaume") {
			mes "[阿塞尔]";
			mes "荣耀属于纪尧姆！";
		} else if (.@team$ == "Croix") {
			mes "[斯万德里]";
			mes "荣耀属于克罗伊！";
		}
		mes "尽情享受这光荣的胜利吧！";
		mes strcharinfo(0)+"，这是胜利的象征。";
	} else {
		set .@amount, .@amount_lose;
		if (.@team$ == "Guillaume") {
			mes "[阿塞尔]";
			mes "虽然战败，但你在战斗中的付出有目共睹。";
			mes "这是纪尧姆·马罗尔对你英勇表现的犒赏！";
			mes "把这次失败当作教训，下次胜利一定属于你。";
		} else if (.@team$ == "Croix") {
			mes "[斯万德里]";
			mes "哦，"+strcharinfo(0)+"，别难过。";
			mes "虽然没能取胜，但我们已经尽力了。";
			mes "这是克罗伊王子的赏赐，请铭记这场战斗。下一场我们一定会赢。";
		}
	}
	close2;

	// If the VIP system is enabled, VIP players receive an extra 2 badges.
	if (VIP_SCRIPT && vip_status(VIP_STATUS_ACTIVE))
		set .@amount, .@amount+2;

	set .@medal_gap, 500 - countitem(.@badge);
	if (.@medal_gap >= .@amount)
		getitem .@badge,.@amount;
	else
		getitem .@badge,.@medal_gap;
	return;
}
