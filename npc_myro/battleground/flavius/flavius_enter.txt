bat_room,133,178,5	script	弗拉维斯军官#01a	418,{
	if (getbattleflag("feature.bgqueue"))
		end;
	if (checkweight(1201,1) == 0) {
		mes "- 等一下！！ -";
		mes "- 你身上带的物品实在太多了。 -";
		mes "- 请减轻一些负重之后 -";
		mes "- 再来试试吧。 -";
		close;
	}
	mes "[纪尧姆军军官]";
	mes "让那些臭克罗伊的家伙见识一下纪尧姆军的厉害！";
	next;
	switch(select("我要加入你们的军队！:结束对话")) {
	case 1:
		if ((BaseJob == Job_Novice) || (BaseJob == Job_Super_Novice)) {
			mes "[纪尧姆军军官]";
			mes "你想加入我们、对抗克罗伊军，我很高兴。但很抱歉：我不能让你这样的新兵上残酷的战场送死。";
		}
		else {
			if (BaseLevel < 80) {
				mes "[纪尧姆军军官]";
				mes "你想加入我们、对抗克罗伊军，我很高兴。但很抱歉：我不能让你这样的新兵上残酷的战场送死。";
			}
			else {
				set .@chk_urtime,checkquest(2070,PLAYTIME);
				if ((.@chk_urtime == 0)|| (.@chk_urtime == 1)) {
					mes "[纪尧姆军军官]";
					mes "你看起来刚从战场回来。";
					mes "现在回去还太早。去休息一下吧，敌人交给我们！";
				}
				else {
					if (.@chk_urtime == 2)
						erasequest 2070;
					set .@mapcount2,getmapusers("bat_b01");
					if (.@mapcount2 > 0) {
						mes "[纪尧姆军军官]";
						mes "我刚收到报告，一支精锐部队已经开赴战场，正在与克罗伊军交战。";
						mes "何不先等一等？在此之前，建议你磨快武器、备好补给。";
					}
					else {
						if ($@FlaviusBG1 > 0) {
							mes "[纪尧姆军军官]";
							mes "已经有一支精锐部队整装待发，准备开赴战场了。";
							mes "何不先等一等？在此之前，建议你磨快武器、备好补给。";
						}
						else {
							mes "[纪尧姆军军官]";
							mes "你看起来已经做好战斗准备了！";
							mes "去让克罗伊的家伙尝尝恐惧的滋味吧！";
							mes "今天，我们的胜利欢呼将响彻整个战场！";
							close2;
							warp "bat_room",85,223;
							end;
						}
					}
				}
			}
		}
		break;
	case 2:
		mes "[纪尧姆军军官]";
		mes "今天，胜利必将属于我们！";
		break;
	}
	close;
}
