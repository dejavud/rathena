	.@item_type = getarg(2);

	mes "[后勤军官]";

	if (!.@item_type)
		mes "你选择了^ff0000"+ getitemname(.@item_id) +"^000000。";
	else if (.@item_type == 1)
		mes "只有一把"+ callfunc("F_GetWeaponType",.@item_id) +"："+ getitemname(.@item_id) +"。";
	else if (.@item_type == 2) {
		mes "你选择了"+ callfunc("F_GetArmorType",.@item_id) +"。";
		mes "80级以上玩家可以装备的这件防具是：^ff0000"+ getitemname(.@item_id) +"^000000。";
	}
	else if (.@item_type == 3) {
		mes "你选择了量产型"+ callfunc("F_GetArmorType",.@item_id) +"。";
		mes "量产型"+ callfunc("F_GetArmorType",.@item_id) +"只有一种：^ff0000"+ getitemname(.@item_id) +"^000000。";
	}
	else if (.@item_type == 4) {
		mes "你选择了普及型"+ callfunc("F_GetArmorType",.@item_id) +"。";
		mes "普及型"+ callfunc("F_GetArmorType",.@item_id) +"只有一种：^ff0000"+ getitemname(.@item_id) +"^000000。";
	}

	if (!.@req_setting)
		mes "这件道具需要^0000ff2,000^000000点KVM点数，"+ .@name$ +"，你的点数是：^580080"+ kvm_point +"^000000。";
	else if (.@req_setting == 4)
		mes "这件道具需要^0000ff1,200^000000点KVM点数，"+ .@name$ +"，你的点数是：^580080"+ kvm_point +"^000000。";
	else
		mes "这件道具需要^0000ff"+ .@prices[.@req_setting] +"^000000点KVM点数，"+ .@name$ +"，你的点数是：^580080"+ kvm_point +"^000000。";

	mes "你确定要这件道具吗？";
	next;
	switch(select("不，我不买了。:是，我要购买。")) {
	case 1:
		mes "[后勤军官]";
		mes "你选择了不购买。";
		mes "购买道具时请多加小心，售出不退。";
		break;
	case 2:
		if (kvm_point >= .@prices[.@req_setting]) {
			kvm_point = kvm_point -.@prices[.@req_setting];
			getitem .@item_id,1;
			mes "[后勤军官]";
			mes "你购买了"+ getitemname(.@item_id) +"。";
			if (!.@req_setting)

			if (.@req_setting == 4)
				mes "你的KVM点数减少了^0000ff1,200^000000点，现在是^580080"+kvm_point+"^000000点。";
			else
				mes "你的KVM点数减少了^0000ff"+ .@prices[.@req_setting] +"^000000点，现在是^580080"+kvm_point+"^000000点。";
		}

			mes "[后勤军官]";
			if (!.@req_setting)
				mes "购买这件道具需要^0000ff2,000^000000点KVM点数。";
			else if (.@req_setting == 4)
				mes "购买这件道具需要^0000ff1,200^000000点KVM点数。";
			else
				mes "购买这件道具需要^0000ff"+ .@prices[.@req_setting] +"^000000点KVM点数。";
