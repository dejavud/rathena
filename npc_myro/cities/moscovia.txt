	mes "在无尽海洋上蔓延的天堂……";
	mes "欢迎来到莫斯科比亚";
	mes "它充满冒险和神秘。";
	next;
	switch(select("关于莫斯科比亚……:去莫斯科比亚。:取消")) {
		case 1:
			mes "[莫斯科比亚 宣传使节]";
			mes "莫斯科比亚是一个美丽的王国，";
			mes "位于卢恩·米德加兹";
			mes "北边的一座岛上。";
			next;
			mes "[莫斯科比亚 宣传使节]";
			mes "我相信你一定会";
			mes "被莫斯科比亚美丽的风景";
			mes "和华丽的宫殿";
			mes "深深迷住。";
			next;
			mes "[莫斯科比亚 宣传使节]";
			mes "现在我们漫长的冬季";
			mes "已经过去了，";
			mes "我很高兴能向你展示";
			mes "我们华丽的家乡。";
			next;
			mes "[莫斯科比亚 宣传使节]";
			mes "如果你想访问莫斯科比亚，";
			mes "就趁现在这个机会！";
			mes "我会帮你";
			mes "愉快地去莫斯科比亚旅行！";
			close;

		case 2:
			mes "[莫斯科比亚 宣传使节]";
			mes "好，那我们现在开始。";
			mes "你应该付我 10,000 Zeny";
			mes "去莫斯科比亚。";
			mes "但你回来时，";
			mes "不用付钱。";
			next;
			mes "[莫斯科比亚 宣传使节]";
			mes "我们现在能出发了吗？";
			next;
			if (select("走吧！:取消") == 2) {
				mes "[莫斯科比亚 宣传使节]";
				mes "如果你现在太忙，";
				mes "请在你想去的时候再告诉我。";
				mes "我随时准备引导任何人去莫斯科比亚。";
				close;
			}
			if (Zeny < 10000) {
				mes "[莫斯科比亚 宣传使节]";
				mes "抱歉，但你现在";
				mes "钱不够";
				mes "你需要 10,000 Zeny";
				mes "才能去莫斯科比亚";
				mes "谢谢。";
				close;
			} else {
				mes "[莫斯科比亚 宣传使节]";
				mes "好，我们现在出发。";
				close2;
				set Zeny,Zeny - 10000;
				warp "moscovia",163,55;
				end;
			}
		case 3:
			mes "[莫斯科比亚 宣传使节]";
			mes "如果你现在太忙，";
			mes "请在你想去的时候再告诉我。";
			mes "我随时准备引导任何人去莫斯科比亚。";
			close;
	}
}
