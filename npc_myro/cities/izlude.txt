	mes "[水手]";
	mes "各位各位！";
	mes "注意啦，注意啦！";
	mes "快来乘风破浪，";
	mes "乘坐迷人的航船吧！";
	mes "快来，快来！";
	next;
	switch(select("柏伊亚岚岛 -> 150 Zeny。:艾尔贝塔码头 -> 500 Zeny。:取消。")) {
	case 1:
		if (Zeny < 150) {
			mes "[水手]";
			mes "150 Zeny！";
			mes "只要 150 Zeny 就能坐船！";
			close;
		}
		set Zeny, Zeny - 150;
		warp "izlu2dun",107,50;
		end;
	case 2:
		if (Zeny < 500) {
			mes "[水手]";
			mes "500 Zeny！";
			mes "只要 500 Zeny 就能坐船！";
			close;
		}
		set Zeny, Zeny - 500;
		warp "alberta",188,169;
		end;
	case 3:
		close;
	}
}
