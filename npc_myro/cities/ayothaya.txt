-	script	::Aibakthing_ayothaya	-1,{
	mes "[哎伊拜婷]";
	mes "呼！哈！呼！哈！";
	mes "让我带你去";
	mes "一个遥远的灵性之地，";
	mes "一片异域神秘的土地，";
	mes "我的家乡阿尤塔亚！";
	next;
	switch(select("关于阿尤塔亚。:去阿尤塔亚。:取消。")) {
	case 1:
		mes "[哎伊拜婷]";
		mes "我们阿尤塔亚人是虔诚的人民，珍视和平与真诚。我们努力过着简单而高尚的生活，与自然和谐相处。";
		next;
		mes "[哎伊拜婷]";
		mes "我们的传统美食举世闻名。当我们端上冬阴功这样的菜肴时毫不奇怪，它由虾、柠檬和辣椒制成，香气和风味诱人得超乎想象。";
		next;
		mes "[哎伊拜婷]";
		mes "对于渴望冒险的人，有一个关于三头虎的古老故事，那是一种出没于我们寺庙遗迹的邪恶生物。";
		next;
		mes "[哎伊拜婷]";
		mes "呼！哈！呼！哈！";
		mes "没有人能抗拒";
		mes "这片美丽土地的召唤。";
		mes "米德加兹人";
		mes "永远受欢迎！";
		close;
	case 2:
		mes "[哎伊拜婷]";
		mes "啊，如果你想访问阿尤塔亚，";
		mes "必须支付 10,000 Zeny。";
		mes "如果你准备好了，我们";
		mes "现在就能出发~";
		next;
		if (select("我准备好了，走吧！:不。") == 1) {
			if (Zeny > 9999) {
				mes "[哎伊拜婷]";
				mes "呼！哈！呼！哈！";
				mes "我们出发吧！回到我";
				mes "美丽的阿尤塔亚！";
				close2;
				set Zeny, Zeny-10000;
				warp "ayothaya",149,71;
				end;
			}
			mes "[哎伊拜婷]";
			mes "抱歉，但你没有 10,000 Zeny 去阿尤塔亚旅行。这样的价格与等待你的体验相比算不了什么！";
			close;
		}
	case 3:
		mes "[哎伊拜婷]";
		mes "你是否梦见过";
		mes "一个充满";
		mes "灵性宁静的美丽地方？你";
		mes "一定梦见的";
		mes "就是阿尤塔亚，我的朋友。";
		close;
	}
}

	mes "[哎伊拜婷]";
	mes "呼！哈！嗯！哈！";
	mes "那么，你觉得阿尤塔亚怎么样？你尝到冬阴功了吗？等你准备好了，我就带你回家。";
	next;
	if (select("回艾尔贝塔。:取消。") == 1) {
		mes "[哎伊拜婷]";
		mes "欢迎你随时回来。希望我们很快能再见面。谢谢~";
		close2;
		if (checkre(0))
			warp "alberta",245,87;
		else
			warp "alberta",235,45;
		end;
	}
	mes "[哎伊拜婷]";
	mes "啊，是的。我明白离开这么美的地方很难。别担心，";
	mes "慢慢来。";
	close;
}
