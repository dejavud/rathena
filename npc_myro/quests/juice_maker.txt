payon_in03,188,146,5	script	马克思·汉森#juice	86,{
	if (checkweight(1201,1) == 0) {
		mes "- 等一下！ -";
		mes "- 你现在身上 -";
		mes "- 带的东西太多了。 -";
		mes "- 请把一些物品存进卡普拉仓库 -";
		mes "- 之后再回来吧。 -";
		close;
	}
	if (MISC_QUEST&1) {
		mes "[商人马克思·汉森]";
		mes "欢迎光临。";
		mes "你也是来";
		mes "加工水果的吗？";
		next;
		switch(select("制作果汁。:聊聊水果加工的事。:取消。")) {
		case 1:
			mes "[商人马克思·汉森]";
			mes "你想做什么果汁？";
			next;
			switch(select("苹果汁:香蕉汁:红萝卜汁:葡萄汁:取消")) {
			case 1:
				set .@fruit,512;
				set .@juice,531;
				break;
			case 2:
				set .@fruit,513;
				set .@juice,532;
				break;
			case 3:
				set .@fruit,515;
				set .@juice,534;
				break;
			case 4:
				set .@fruit,514;
				set .@juice,533;
				break;
			case 5:
				mes "[商人马克思·汉森]";
				mes "那么……";
				mes "下次再见。";
				close;
			}
			if (countitem(.@fruit) == 0 || countitem(713) == 0 || Zeny < 3) {
				mes "[商人马克思·汉森]";
				mes "哦，天哪……";
				mes "你还没带齐所需的材料。要做"+getitemname(.@juice)+"，我需要1个"+getitemname(.@fruit)+"和1个空瓶。另外还要收3 zeny的手续费。";
				next;
				mes "[商人马克思·汉森]";
				mes "等你把东西都准备好，我就会把水果榨成美味的果汁给你。";
				close;
			}

			mes "[商人马克思·汉森]";
			mes "你想做多少？";
			next;
			switch(select("能做多少就做多少。:我要指定数量。:取消。")) {
			case 1:
				set .@make,countitem(.@fruit);
				if (countitem(713) < .@make) set .@make, countitem(713);
				if (Zeny/3 < .@make) set .@make, Zeny/3;
				break;
			case 2:
				mes "[商人马克思·汉森]";
				mes "请输入100以内的数字。不想做的话就输入『0』。你最多可以做" + countitem(.@fruit) + "瓶果汁。";
				next;
				while(1) {
					input .@input;
					if (.@input == 0) {
						mes "[商人马克思·汉森]";
						mes "那么……";
						mes "欢迎再来。";
						close;
					}
					else if (.@input > 100) {
						mes "[商人马克思·汉森]";
						mes "超过100瓶可做不了。请换个数量。";
						next;
					}
					else {
						break;
					}
				}
				set .@make,.@input;
				break;
			case 3:
				mes "[商人马克思·汉森]";
				mes "那么……";
				mes "欢迎再来。";
				close;
			}

			set .@total_zeny,3 * .@make;

			if (countitem(.@fruit) < .@make || countitem(713) < .@make || Zeny < .@total_zeny) {
				mes "[商人马克思·汉森]";
				mes "哦，天哪……";
				mes "你还没带齐所需的材料。这种情况我也无能为力。";
				mes "看来你得再去收集一下了。";
				close;
			}
			delitem .@fruit,.@make;
			delitem 713,.@make;
			set Zeny, Zeny-.@total_zeny;
			getitem .@juice,.@make;

			mes "[商人马克思·汉森]";
			mes "拿好了！如约定的那样，新鲜又美味的果汁。喝起来一定很爽口。";
			next;
			mes "[商人马克思·汉森]";
			mes "那么……";
			mes "欢迎再来。";
			close;
		case 2:
			mes "[商人马克思·汉森]";
			mes "在人类发展出如此庞大的社会之前，";
			mes "是靠采集树上的果实生存的。";
			mes "水果也许是大自然让我们得以存在于世的恩赐。";
			next;
			mes "[商人马克思·汉森]";
			mes "自从生活富裕起来，年轻人似乎都不吃水果了。于是，";
			mes "我开始琢磨让水果变得更容易入口的办法。";
			next;
			mes "[商人马克思·汉森]";
			mes "我发现把水果做成果汁，不但方便，味道也好得多。";
			next;
			mes "# 果汁信息 #";
			mes "^CC4E5C- 苹果汁 -^000000";
			mes "苹果1个，空瓶1个，3 zeny。";
			mes "^E3CF57- 香蕉汁 -^000000";
			mes "香蕉1个，空瓶1个，3 zeny。";
			mes "^ED9121- 红萝卜汁 -^000000";
			mes "红萝卜1个，空瓶1个，3 zeny。";
			mes "^CC00FF- 葡萄汁 -^000000";
			mes "葡萄1个，空瓶1个，3 zeny。";
			close;
		case 3:
			mes "[商人马克思·汉森]";
			mes "喂！";
			mes "拜访别人的时候，";
			mes "总得打声招呼吧！";
			close;
		}
	}
	else {
		mes "[商人马克思·汉森]";
		mes "欢迎光临。";
		mes "有什么可以为你效劳的？";
		next;
		if (select("聊天:取消") == 1) {
			mes "[商人马克思·汉森]";
			mes "在人类发展出如此庞大的社会之前，";
			mes "是靠采集树上的果实生存的。";
			mes "水果也许是大自然让我们得以存在于世的恩赐。";
			next;
			mes "[商人马克思·汉森]";
			mes "自从生活富裕起来，年轻人似乎都不吃水果了。于是，";
			mes "我开始琢磨让水果变得更容易入口的办法。";
			next;
			mes "[商人马克思·汉森]";
			mes "我发现把水果做成果汁，不但方便，味道也好得多。";
			close;
		}
		mes "[商人马克思·汉森]";
		mes "喂！";
		mes "拜访别人的时候，";
		mes "总得打声招呼吧！";
		close;
	}
}
