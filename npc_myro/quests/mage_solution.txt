pay_arche,122,100,0	script	多尔肖	88,{
	mes "[魔法师公会人员]";
	mes "想要溶液吗？嗯，给我50 Zeny和至少1个空试管。";
	next;
	if (select("好，成交。:算了，不要了。") == 1) {
		mes "[魔法师公会人员]";
		if (Zeny < 50) {
			mes "喂！你的钱不够付我50 Zeny的费用。";
			close;
		}
		if (countitem(1092) == 0) {
			mes "没有瓶子可装不了溶液！带一个空试管来给我。";
			close;
		}
		delitem 1092,1; //Empty_Cylinder
		set Zeny, Zeny-50;
		getitem 1089,1; //Payon_Potion
	}
	close;
}

	}
	close;
}
