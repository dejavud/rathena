		mes "它用一块厚厚的金属板护住你的脸。说真的，";
		mes "这是人人都该想要的东西！就算你自己用不上，";
		mes "送给正在学锻造的朋友也不错。";
		next;
		mes "[铁匠]";
		mes "怎么样？只要给我^2244FF50个钢铁^000000和^4422FF仅仅2000 Zeny^000000，我马上就为你做出来～！！";
		next;
		switch(select("嗯……不错。好吧。:它到底有什么用？")) {

			if ((countitem(999) > 49) && (Zeny > 1999)){
				mes "[铁匠]";
				mes "啊，看得出你懂得欣赏铁匠的作品！好，我马上就做！！";

				mes "[铁匠]";
				mes "哈哈哈～！好好利用它吧！";
				close;
			}
			else {
				mes "[铁匠]";
				mes "嗯……";
				mes "你还没带齐我要的东西。";
				next;
				mes "[铁匠]";
				mes "普通铁匠会觉得这是对他们的不敬，但我会原谅你。";
				mes "想让我做这个面具的话，别忘了我需要什么。";
				close;
			}
		case 2:
			mes "[铁匠]";
			mes "^3355FF钢铁面具^000000用一块厚厚的金属板护住你的脸。";
			mes "这个发明是我在锤子和铁砧前辛劳30年的结晶。";
			close;
		}
	case 2:
		mes "[铁匠]";
		mes "啊……！";
		mes "看来你是第一次见到我这么了不起的铁匠，";
		mes "惊讶得说不出话来了吧。请便，等你准备好了再开口。";
		next;
		mes "^3355FF这家伙似乎有点自以为是。^000000";
		close;
	}
}

// Number Match Game :: gef_yagu
//============================================================
geffen,122,77,4	script	妮亚#yagu	700,{
	mes "[妮亚]";
	mes "你好，我是妮亚，游戏大师";
	mes "马翁的第五个弟子。";
	mes "怎么样，要不要跟我玩一局";
	mes "猜数字游戏？";
	next;
	switch(select("猜数字游戏?:练习猜数字:玩猜数字:纪录保持者:取消")) {
	case 1:
		mes "[妮亚]";
		mes "啊，这个嘛，猜数字游戏的";
		mes "目标就是猜出我心里想的";
		mes "数字。当然，我能选的";
		mes "数字有一些限制规则。";
		next;
		mes "[妮亚]";
		mes "我想的数字必须是3位数，";
		mes "任何一位都不能是0，";
		mes "而且每一位的数字";
		mes "都不能重复。";
		next;
		mes "[妮亚]";
		mes "比如说，我不能选";
		mes "103、112、252或者701。";
		mes "这些数字要么含有0，";
		mes "要么有重复的数字。";
		mes "现在明白了吧？";
		next;
		mes "[妮亚]";
		mes "你只有5次猜测的机会，";
		mes "每次猜完，我会告诉你";
		mes "猜对了几个数字，";
		mes "以及有几个数字不仅猜对，";
		mes "而且位置也猜对了。";
		next;
		mes "[妮亚]";
		mes "举个例子，假设";
		mes "我想的数字是168。";
		mes "如果你第一次猜678，";
		mes "那你就猜对了6和8两个数字，";
		mes "不过只有8的位置是对的。";
		next;
		mes "[妮亚]";
		mes "对了，建议你先用";
		mes "练习模式试试。";
		mes "等你熟悉了，再来玩";
		mes "正式的猜数字游戏。";
		mes "别忘了，速度也算进成绩的！";
		next;
		mes "[妮亚]";
		mes "啊，如果你在猜数字游戏里";
		mes "创造了新纪录，";
		mes "你的名字就会被记录下来，";
		mes "所以千万别错过这个";
		mes "成名、甚至发财的机会～";
		close;
	case 2:
		mes "[妮亚]";
		mes "好，那我们来玩一局";
		mes "练习模式的猜数字。";
		mes "请试着猜出我的数字，好吗？";
		mes "如果想取消，请输入一个大于1000的数字。";
		set .@practice,1;
		break;
	case 3:
		mes "[妮亚]";
		mes "好，我们来玩猜数字吧！";
		mes "你有5次机会猜出我心里想的数字。";
		mes "如果想取消，请输入一个大于1000的数字。";
		set .@practice,0;
		break;
	case 4:
		set .@name_record$, $050908_minus1_yagu$;
		if (.@name_record$ == "") //custom
			set .@name_record$,"(null)";
		set .@score_record, $050908_minus1_yagu;
		set .@score_min, .@score_record / 60;
		set .@score_sec, .@score_record % 60;
		mes "[妮亚]";
		mes "^ff0000" + .@name_record$ + "^000000";
		mes "是猜数字游戏的纪录保持者，";
		mes "成绩是";
		mes "^ff0000" + .@score_min + "分" + .@score_sec + "秒^000000。";
		close;
	case 5:
		mes "[妮亚]";
		mes "保重了，冒险者。";
		mes "要是你想玩猜数字游戏，";
		mes "随时来找我，好吗？";
		mes "再见～";
		close;
	}
	next;

	// Generate 3 digits of a number.
	while(1) {
		set .@yagu100, rand(1,9);
		set .@yagu10, rand(1,9);
		set .@yagu1, rand(1,9);
		if ((.@yagu100 != .@yagu10) && (.@yagu100 != .@yagu1) && (.@yagu10 != .@yagu1)) {
			// Store the actual number (unofficial).
			set .@yagu, (100*.@yagu100) + (10*.@yagu10) + .@yagu1;
			break;
		}
	}

	// The official time calculation is based on ServerTime,
	// an int HHMMSS equivalent to atoi(gettimestr("%H%M%S",7)).
	// UNIX time is really a better way to do this.
	set .@start_time, gettimetick(2);

	while(1) {

		// Guess a number that meets all conditions.
		while(1) {
			if (input(.@input,100,999) != 0) {
				mes "[妮亚]";
				mes "你取消了";
				mes "猜数字游戏。";
				close;
			}
			set .@input100, .@input / 100;
			set .@input10, (.@input % 100) / 10;
			set .@input1, .@input % 10;
			if ((.@input100 > 0) && (.@input10 > 0) && (.@input1 > 0)) {
				if ((.@input100 != .@input10) && (.@input100 != .@input1) && (.@input10 != .@input1))
					break;
			}
			mes "[妮亚]";
			mes "你把某一位输成了0，";
			mes "或者有重复的数字。";
			mes "请再猜一次。";
			next;
		}

		set .@retry, .@retry+1;
		mes "[妮亚]";
		mes "^ff0000"+strcharinfo(0)+"^000000，";
		mes "你猜的是^0000ff"+.@input+"^000000。";
		mes "稍等一下，";
		mes "我来算算结果。";
		next;

		// Correct digits in correct locations.
		set .@strike,0;
		if (.@yagu100 == .@input100)
			set .@strike, .@strike+1;
		if (.@yagu10 == .@input10)
			set .@strike, .@strike+1;
		if (.@yagu1 == .@input1)
			set .@strike, .@strike+1;

		// Correct digits in incorrect locations.
		set .@ball,0;
		if ((.@yagu100 == .@input10) || (.@yagu100 == .@input1))
			set .@ball, .@ball+1;
		if ((.@yagu10 == .@input100) || (.@yagu10 == .@input1))
			set .@ball, .@ball+1;
		if ((.@yagu1 == .@input100) || (.@yagu1 == .@input10))
			set .@ball, .@ball+1;

		if (.@strike == 3) {
			if (.@practice) {
				mes "[妮亚]";
				mes "对，我想的数字";
				mes "就是^ff0000"+.@yagu+"^000000！";
				mes "恭喜你赢得了";
				mes "这局练习赛！";
				mes "现在你应该可以";
				mes "挑战正式的了～";
				close;
			} else {

				// Calculate final time and check against the previous record.
				set .@end_time, gettimetick(2);
				if (.@end_time < .@start_time)
					set .@end_time, .@end_time + 3600;
				set .@total_time, .@end_time - .@start_time;
				set .@total_min, .@total_time / 60;
				set .@total_sec, .@total_time % 60;
				set .@name_record$, $050908_minus1_yagu$;
				if (.@name_record$ == "") //custom
					set .@name_record$,"(null)";
				set .@score_record, $050908_minus1_yagu;
				set .@score_min, .@score_record / 60;
				set .@score_sec, .@score_record % 60;

				if (.@score_record >= .@total_time || !.@score_record) { //custom (second condition)
					set $050908_minus1_yagu$, strcharinfo(0);
					set $050908_minus1_yagu, .@total_time;
					mes "[妮亚]";
					mes "太棒了！你猜出了";
					mes "我的数字，就是";
					mes "^ff0000" + .@yagu + "^000000。你的成绩是";
					mes "^ff0000" + .@total_min + "分" + .@total_sec + "秒^000000，";
					mes "打破了原先";
					mes "^ff0000" + .@score_min + "分" + .@score_sec + "秒^000000的纪录。";
					next;
					mes "[妮亚]";
					mes "旧纪录是";
					mes "^ff0000" + .@name_record$ + "^000000";
					mes "创下的，不过现在";
					mes "你就是猜数字游戏";
					mes "新的擂主了。";
					mes "恭喜你创造新纪录！";
					close;
				}
				mes "[妮亚]";
				mes "干得漂亮，你猜出了";
				mes "我的数字！答案当然就是";
				mes "^ff0000" + .@yagu + "^000000。你总共用了";
				mes "^ff0000" + .@total_min + "分" + .@total_sec + "秒^000000";
				mes "就猜对了这个数字。";
				next;
				mes "[妮亚]";
				mes "^ff0000" + .@name_record$ + "^000000";
				mes "是目前猜数字游戏的";
				mes "纪录保持者，成绩是";
				mes "^ff0000" + .@score_min + "分" + .@score_sec + "秒^000000。";
				mes "祝你好运，希望你有一天";
				mes "能打破这个纪录～";
				close;
			}
		} else {
			if (.@practice) {
				mes "[妮亚]";
				mes "你猜的数字里有^ff0000" + .@ball + "^000000个";
				mes "数字是对的。其中";
				mes "^ff0000" + .@strike + "^000000个不仅数字对，";
				mes "位置也是对的。";
				mes "好，再试一次～";
				next;
			} else {
				mes "[妮亚]";
				mes "你上次猜的数字里有";
				mes "^ff0000" + .@ball + "^000000个数字是对的，";
				mes "其中^ff0000" + .@strike + "^000000个不仅数字对，";
				mes "位置也是对的。";
				switch(.@retry) {
					case 1: mes "你还有四次机会。"; break;
					case 2: mes "你还有三次机会。"; break;
					case 3: mes "你还有两次机会。"; break;
					case 4: mes "你只剩最后一次机会了……"; break;
					case 5: mes "答案是^ff0000" + .@yagu + "^000000。"; close;
				}
				next;
			}
		}
	}
}
