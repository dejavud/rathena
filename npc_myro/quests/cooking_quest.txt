S_SellSets:
	.@item_id = getarg(0);
	.@item_cost = getiteminfo(.@item_id, ITEMINFO_BUY);
	.@item_weight = getiteminfo(.@item_id, ITEMINFO_WEIGHT);
	mes "[玛德琳·朱]";
	mes "你要购买多少个"+ (.@item_id == 12125 ? "户外":"家庭式");
	mes "料理器具？如果想取消，请输入数字 0。";
	next;
	while(1) {
		input .@sell;
		if (.@sell == 0) {
			mes "[玛德琳·朱]";
			mes "你改变主意了？";
			mes "好吧，如果之后你需要";
			mes "购买"+ getitemname(.@item_id);
			mes "的话，随时回来找我，好吗？";
			close;
		}
		else if (.@sell > 100) {
			mes "[玛德琳·朱]";
			mes "哦，抱歉，为了保险起见，";
			mes "一次最多只出售 100 个"+ getitemname(.@item_id) +"。";
			next;
		}
		else break;
	}
	set .@total_cost,.@sell * .@item_cost;
	set .@total_weight,.@sell * .@item_weight;
	if (Zeny < .@total_cost) {
		mes "[玛德琳·朱]";
		mes "哦，抱歉，你的钱";
		mes "不够买这么多";
		mes getitemname(.@item_id);
		mes "购买前请先确认";
		mes "你的 zeny 够不够~";
		close;
	}
	if (!checkweight(.@item_id,.@sell)) {
		mes "[玛德琳·朱]";
		mes "抱歉，你的物品栏没有足够的空间";
		mes "装下这么多";
		mes getitemname(.@item_id) +"……";
		close;
	}
	set Zeny, Zeny-.@total_cost;
	getitem .@item_id,.@sell;
	mes "[玛德琳·朱]";
	mes "给你~";
	mes "祝你烹饪修行顺利！";
	return;
}
