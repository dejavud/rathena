alberta,26,229,0	script	卡普拉服务人员#bunny	83,{
	if (BUNYBND == 1) {
		mes "[卡普拉服务人员]";
		mes "你好～！";
		mes "你想参加";
		mes "卡普拉公司的特别活动";
		mes "^529DFF兔耳发圈活动^000000";
		mes "吗？";
		next;
		switch(select("当然，我带来物品了。:活动信息:取消")) {
		case 1:
			mes "[卡普拉服务人员]";
			mes "好的～";
			mes "让我检查一下";
			mes "你是不是把物品";
			mes "都带来了……";
			next;
			setarray .@Items[0],949,100,706,1,722,1,2213,1;
			for(set .@i,0; .@i<8; set .@i,.@i+2)
				if (countitem(.@Items[.@i]) < .@Items[.@i+1]) {
					mes "[卡普拉服务人员]";
					mes "哦，很抱歉，";
					mes "但你至少还需要";
					mes "带来";
					mes callfunc("F_InsertPlural",.@Items[.@i+1],getitemname(.@Items[.@i]))+"。";
					close;
				}
			mes "[卡普拉服务人员]";
			mes "太好了，看来你已经";
			mes "收集齐了制作";
			mes "兔耳发圈所需的";
			mes "全部物品。";
			mes "请稍等一下，";
			mes "我这就把它做出来……";
			next;
			for(set .@i,0; .@i<8; set .@i,.@i+2)
				if (countitem(.@Items[.@i]) < .@Items[.@i+1]) {
					mes "[卡普拉服务人员]";
					mes "嗯？抱歉，";
					mes "我现在还做不了。";
					mes "要做出这条兔耳发圈，";
					mes "你还需要"+callfunc("F_InsertPlural",.@Items[.@i+1],getitemname(.@Items[.@i]));
					mes "才行……";
					close;
				}
			delitem 949,100; //Feather
			delitem 706,1; //Four_Leaf_Clover
			delitem 722,1; //Scarlet_Jewel
			delitem 2213,1; //Cat_Hairband
			mes "[卡普拉服务人员]";
			mes "啊，做好了～";
			mes "完美的兔耳发圈！";
			mes "希望你会喜欢。";
			getitem 2214,1; //Bunny_Band
			set BUNYBND,0;
			next;
			mes "[卡普拉服务人员]";
			mes "感谢你参加这次";
			mes "特别活动，也感谢你";
			mes "一直以来对卡普拉";
			mes "服务的支持～";
			close;
		case 2:
			goto L_Info;
		case 3:
			close;
		}
	}
	mes "[卡普拉服务人员]";
	mes "你好～！你想参加由卡普拉";
	mes "公司主办、艾尔贝塔商人公会";
	mes "赞助的特别活动";
	mes "^529DFF兔耳发圈活动^000000";
	mes "吗？";
	next;
	switch(select("参加活动:活动信息:取消")) {
	case 1:
		set BUNYBND,1;
		mes "[卡普拉服务人员]";
		mes "太好了！感谢你的";
		mes "参与！如果你还没";
		mes "听说过的话：想让我";
		mes "帮你做一条兔耳发圈，";
		mes "你需要收集";
		mes "这些物品……";
		next;
		mes "[卡普拉服务人员]";
		mes "100根柔毛、";
		mes "1棵四叶幸运草、";
		mes "1个猫耳发圈";
		mes "和1颗珍珠。就这些！";
		mes "祝你好运，";
		mes "我会在这里等你～";
		close;
	case 2:
		break;
	case 3:
		close;
	}
	L_Info:
	mes "[卡普拉服务人员]";
	mes "为了感谢尊贵的顾客，";
	mes "卡普拉公司准备了";
	mes "一项特别活动：";
	mes "只要冒险者带来所需物品，";
	mes "卡普拉服务人员就会";
	mes "为他制作兔耳发圈。";
	next;
	mes "[卡普拉服务人员]";
	mes "想获得这件独一无二的";
	mes "特别物品，";
	mes "请带来100根柔毛、";
	mes "1棵四叶幸运草、";
	mes "1个猫耳发圈";
	mes "和1颗珍珠。";
	next;
	mes "[卡普拉服务人员]";
	mes "准备好之后，";
	mes "请带着那些物品";
	mes "回来找我。";
	mes "（由艾尔贝塔";
	mes "商人公会赞助）";
	close;
}
