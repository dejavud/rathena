// Edgar's Offer :: iz_npc
//============================================================
-	script	::Edgar_izlude	-1,{
	if (MISC_QUEST & 16) {
		mes "[艾德加]";
		mes "你又要去艾尔贝塔吗？那就跟上次一样，给你同样的优惠，只收250 Zeny。";
		mes "怎么样？";
		next;
		if (select("好呀～！:你为什么对我这么好？！") == 1) {
			if (Zeny < 250) {
				mes "[艾德加]";
				mes "嗯……";
				mes "这点钱";
				mes "不够啊。";
				mes "你何不再去";
				mes "弄点钱来？";
				close ;
			}
			set Zeny, Zeny-250;
			warp "alberta",195,164;
			end;
		}
		mes "[艾德加]";
		mes "我就是这样的人。再说，你那狡黠的笑容让我想起我金发的儿子，";
		mes "他多年前就离家出海，在自己的船上当了水手。愿他平安，";
		mes "不管他在哪里。";
		close;
	}
	mes "[艾德加]";
	mes "我的家乡依斯鲁德，通过西边的港口跟艾尔贝塔相连。";
	mes "两地之间往来频繁，我简直快变成艾尔贝塔人了。呵呵呵～";
	next;
	mes "[艾德加]";
	mes "我认识一个住在艾尔贝塔的家伙，叫菲历斯，我们挺熟的。";
	mes "那家伙真是小气……干什么都要收钱！";
	next;
	mes "[艾德加]";
	mes "不过他其实是个好人，喜欢帮助别人。他心地善良，";
	mes "只要你出得起价，他就会帮你。";
	next;
	mes "[艾德加]";
	mes "最近艾尔贝塔的人都说，他是真心想帮大家，";
	mes "要杰勒比结晶只是个幌子。嗯，你该去看看他能提供什么。";
	next;
	if (select("能告诉我去艾尔贝塔怎么走吗？:结束对话") == 1) {
		mes "[艾德加]";
		mes "啊？嗯，你可以用两条腿走着去。不过要是有钱的话，我建议你坐船。";
		next;
		if (select("好，知道了。:可我走累了，又没钱！") == 1) {
			mes "[艾德加]";
			mes "好的，保重～";
			close;
		}
		mes "[艾德加]";
		mes "好吧……";
		mes "你不想走路，又没钱，可还是想去那儿？哦，天哪……";
		next;
		mes "[艾德加]";
		mes "行吧行吧。我呢，身为船长，可以便宜点捎你过去。250 Zeny怎么样？";
		next;
		if (select("好呀～！:呸，真黑！！") == 1) {
			set MISC_QUEST,MISC_QUEST | 16;
			if (Zeny < 250) {
				mes "[艾德加]";
				mes "嗯……";
				mes "这点钱";
				mes "不够啊。";
				mes "再去";
				mes "弄点来吧。";
				close;
			}
			set Zeny, Zeny-250;
			warp "alberta",195,164;
			end;
		}
		mes "[艾德加]";
		mes "哎呀呀，";
		mes "要是你觉得";
		mes "这都算黑的话……";
		close;
	}
	mes "[艾德加]";
	mes "嗯，好吧。";
	mes "回见～";
	close;
}
