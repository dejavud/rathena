function	script	Func_Are_Rew	{
	.@item_id = getarg(0);
	.@amount = getarg(1);
	.@arena_point = getarg(2);

	mes "[吉武]";
	mes "您想用竞技场积分兑换 "+ getitemname(.@item_id) +" 吗？";
	mes "可以用 ^3131FF"+ .@arena_point +" 点竞技场积分兑换 "+ .@amount +" 个 "+ getitemname(.@item_id) +"^000000。";
	mes "如果想取消，请输入 0。如果不想取消，请输入您想花费的竞技场积分的 ^3131FF倍数^000000。";
	next;
	input .@reward;
	if (.@reward <= 0) {
		mes "[吉武]";
		mes "您取消了请求。";
		close;
	} else if (.@reward > 1500) {
		mes "[吉武]";
		mes "您超出了最大上限。";
		close;
	}
	if (arena_point < (.@reward * .@arena_point)) {
		mes "[吉武]";
		mes "您的竞技场积分不够。请确认您拥有的竞技场积分总数。";
		close;
	}
	arena_point = arena_point - (.@reward * .@arena_point);
	getitem .@item_id,.@reward * .@amount;
	mes "[吉武]";
	mes "谢谢，欢迎再来。";
	close;
}
