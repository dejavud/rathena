//===== Description: ========================================= 
//= [Official Conversion]
//= 积分兑换：竞速赛道积分 -> 竞技场积分
//===== Additional Comments: =================================
//= 1.0 First version. [L0ne_W0lf]
//============================================================

prt_are_in,103,11,3	script	竞技场积分管理员	732,{
	mes "[竞技场积分管理员]";
	mes "希望您在竞技场过得愉快。如果您在";
	mes "艾尔帕兰的竞速赛道赢得了竞速赛道积分，我可以";
	mes "把它们转换成竞技场积分。";
	next;
	switch(select("查询积分:转换积分:^660000转换说明^000000")) {
	case 1:
		mes "[竞技场积分管理员]";
		mes "" + strcharinfo(0) + "，您目前有";
		mes "" + arena_point + " 点竞技场积分";
		mes "和 " + tt_point + " 点竞速赛道积分。";
		close;
	case 2:
		mes "[竞技场积分管理员]";
		mes "" + strcharinfo(0) + "，您目前有";
		mes "" + arena_point + " 点竞技场积分";
		mes "和 " + tt_point + " 点竞速赛道积分。";
		next;
		mes "[竞技场积分管理员]";
		mes "请选择赛道积分转竞技场积分的兑换方式。请注意，";
		mes "一次转换超过 10 点";
		mes "赛道积分时，只能以 ^4D4DFF10 的倍数^000000 转换。";
		next;
		switch(select("2 TP -> 1 AP:4 TP -> 2 AP:6 TP -> 3 AP:8 TP -> 4 AP:10 TP 及以上:取消")) {
		case 1:
			callsub S_ExchangePoints,28999,2,1;
		case 2:
			callsub S_ExchangePoints,28998,4,2;
		case 3:
			callsub S_ExchangePoints,28997,6,2;
		case 4:
			callsub S_ExchangePoints,28996,8,4;
		case 5:
			mes "[竞技场积分管理员]";
			mes "请输入您想把10 点竞速赛道积分转换成";
			mes "竞技场积分的次数。可输入的最大值是 20。";
			mes "要取消，请输入 ''^3355FF0^000000''。";
			next;
			input .@input;
			if (.@input <= 0) {
				mes "[竞技场积分管理员]";
				mes "您已取消本项服务。";
				close;
			}
			else if (.@input > 20) {
				mes "[竞技场积分管理员]";
				mes "您的请求超出了最大上限。请输入不超过 20 的数值。";
				close;
			}
			else {
				set .@want_point1,10 * .@input;
				set .@want_point,5 * .@input;
				set .@my_arena_all,arena_point + .@want_point;
				set .@my_turbo_all,tt_point - .@want_point1;
				if (.@my_arena_all > 28999) {
					mes "[竞技场积分管理员]";
					mes "如果继续转换您的竞速赛道积分，竞技场积分将超过上限。";
					mes "竞技场积分不能超过 29,000 点。";
					next;
					mes "[竞技场积分管理员]";
					mes "请在使用本服务前先花掉一些竞技场积分。感谢您的惠顾。";
					close;
				}
				if (.@my_turbo_all < 0) {
					mes "[竞技场积分管理员]";
					mes "抱歉，您的竞速赛道积分不足以进行这次竞技场积分转换。";
					close;
				}
				else {
					mes "[竞技场积分管理员]";
					mes "您已把10 点竞速赛道积分";
					mes "转换成 5 点竞技场积分，共 " + .@input + " 次。";
					mes "总共 " + .@want_point1 + " 点竞速赛道积分被转换成";
					mes "" + .@want_point +" 点竞技场积分。";
					set tt_point, .@my_turbo_all;
					set arena_point, .@my_arena_all;
					next;
					mes "[竞技场积分管理员]";
					mes "" + strcharinfo(0) + "，您现在有";
					mes "^4682B4" + arena_point + "^000000 点竞技场积分";
					mes "和 ^00688B" + tt_point + "^000000 点竞速赛道积分。感谢您的惠顾。";
					close;
				}
			}
		case 6:
			mes "[竞技场积分管理员]";
			mes "您已取消本项服务。";
			close;
		}
	case 3:
		// 同样。这个选项没有对话。
		close;
	}
	close;

S_ExchangePoints:
	if (arena_point > getarg(0) ) {
		mes "[竞技场积分管理员]";
		mes "如果继续转换您的竞速赛道积分，竞技场积分将超过上限。";
		mes "竞技场积分不能超过 29,000 点。";
		next;
		mes "[竞技场积分管理员]";
		mes "请在使用本服务前先花掉一些竞技场积分。感谢您的惠顾。";
		close;
	}
	.@track_point = getarg(1);
	if (tt_point >= .@track_point) {
		.@arena_point = getarg(2);
		mes "[竞技场积分管理员]";
		mes "" + strcharinfo(0) + "，";
		mes "您已把 "+ .@track_point +" 点赛道";
		mes "积分转换成 "+ .@arena_point +" 点竞技场积分。";
		tt_point = tt_point - .@track_point;
		arena_point = arena_point + .@arena_point;
		next;
		mes "[竞技场积分管理员]";
		mes "" + strcharinfo(0) + "，您现在有";
		mes "^4682B4" + arena_point + "^000000 点竞技场积分";
		mes "和 ^00688B" + tt_point + "^000000 点竞速赛道积分。感谢您的惠顾。";
		close;
	}
	else {
		mes "[竞技场积分管理员]";
		mes "抱歉，您的竞速赛道积分不够。";
		mes "本转换服务至少需要 2 点竞速赛道积分。";
		close;
	}
}
