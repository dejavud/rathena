		mes "你轻轻松松地举起了石头。^000000";

		mes "哇~ 太棒了~下次见面，我带你去个好地方。回见~";

		mes "对你来说似乎太重了。^000000";
		next;
		mes "[宋武斗]";
		mes "你还缺乏锻炼。多练练再来吧。";
	}
	return;
}
