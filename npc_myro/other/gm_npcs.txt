// callfunc("F_GM_NPC"{,<password>,<0:int | 1:str>{,<min>,<max>}})
function	script	F_GM_NPC	{

	// 要禁用所有 GM NPC，取消下面一行的注释。
	//end;

	// 要设置访问这些 NPC 所需的最低 GM 等级，请修改下面一行。
	if (getgmlevel() < 99) {
		// 记录事件。
		getmapxy(.@map$,.@x,.@y,BL_NPC);
		logmes strcharinfo(0)+" attempted to access GM NPC "+strnpcinfo(0)+" ("+.@map$+","+.@x+","+.@y+").";
		end;
	}

	if (!getargcount()) return;

	// 要禁用密码输入，取消下面一行的注释。
	//return 1;

	// 密码返回值：
	//  1: 输入正确
	//  0: 输入错误
	// -1: 输入为零/取消（仅限整数）
	// -2: 超出范围（仅限整数）
	if (getarg(1) == 0) {
		// 整数密码
		if (getarg(3,0)) {
			.@error = input(.@input,getarg(2),getarg(3));
			if (.@error)
				return -2;
		} else
			input .@input;
		if (.@input == 0)
			return -1;
		else
			return (.@input == getarg(0))?1:0;
	} else {
		// 字符串密码
		input .@inputstr$;
		return (.@inputstr$ == getarg(0))?1:0;
	}
}
