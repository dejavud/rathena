				mes "很抱歉，你的基础等级还不够高，无法雇用该佣兵。";

-	script	佣兵商人#dummy::MercMerchant	-1,{
	mes "[佣兵道具商人]";
	mes "你好，我出售佣兵可以使用的道具。有没有你特别想";
	mes "寻找的东西？";
	next;
	setarray .@item, 12184,	12185,	12241,	12242,	12243;
	setarray .@cost, 2500,	5000,	800,	1500,	3000;
	for (.@size = getarraysize(.@item); .@i < .@size; .@i++)
		.@menu$ = .@menu$ + getitemname(.@item[.@i]) + ":";
	.@m = select(.@menu$)-1;
	.@str$ = .@cost[.@m] + "";
	.@len = getstrlen(.@str$);
	.@str_cost$ = (.@len <= 3)? .@str$ : insertchar(.@str$,",",.@len-3);
	mes "[佣兵道具商人]";
	mes getitemname(.@item[.@m]);
	mes "每个售价 " + .@str_cost$ + " zeny。你要多少个？";
	next;
	input .@input;
	if (!.@input) {
		mes "[佣兵道具商人]";
		mes "你改变主意了？好，想买佣兵药水时随时欢迎你再来找我。";
		close;
	}
	else if (.@input < 0 || .@input > 10000) {
		mes "[佣兵道具商人]";
		mes "很抱歉，你的一次最多只能购买 10,000 个。请输入";
		mes "1 到 10,000 之间的数字。";
		close;
	}
	else if (set(.@total_po, .@input * .@cost[.@m]) > Zeny) {
		mes "[佣兵道具商人]";
		mes "很抱歉，你的zeny 不够买这么多药水。嗯，等你准备好";
		mes "为你的佣兵购买道具时，我会在这里等你。";
		close;
	}
	else if (!checkweight(.@item[.@m], .@input)) {
		mes "[佣兵道具商人]";
		mes "如果给你这么多药水，你会拿不动的。请在腾出";
		mes "一些物品栏空间后再来。";
		close;
	}
	mes "[佣兵道具商人]";
	if (.@input == 1) 
		mes "这是你要的 " + getitemname(.@item[.@m]) + "。";
	else {
		mes "给你，一共是";
		mes callfunc("F_InsertPlural",.@input,getitemname(.@item[.@m])) + "。";
	}
	mes "谢谢，需要更多佣兵药水时请再来。";
	Zeny = Zeny - .@total_po;
	getitem .@item[.@m], .@input;
	close;
}

-	script	Mercenary_Switch	-1,{

	setarray .@name$, "枪兵","剑兵","弓兵";
	.@npc$ = strnpcinfo(2);
	for (.@size = getarraysize(.@name$); .@type < .@size; .@type++)
		if (.@npc$ == .@name$[.@type]) break;

	mes "[检查员]";
	mes "请输入密码。";
	next;
	.@i = callfunc("F_GM_NPC",1854,0,0,10000);
	mes "[检查员]";
	if (.@i == -2)
		mes "错误。";
	else if (.@i == 0)
		mes "数字错误";
	else if (.@i == 1) {
		mes "请选择。";
		next;
		switch( select( "关闭佣兵 NPC","开启佣兵 NPC" ) ) {
		case 1:
			mes "NPC 已关闭。";
			disablenpc "佣兵管理员#" + .@name$[.@type];
			disablenpc "佣兵商人#" + .@name$[.@type];
			break;
		case 2:
			mes "NPC 已开启。";
			enablenpc "佣兵管理员#" + .@name$[.@type];
			enablenpc "佣兵商人#" + .@name$[.@type];
			break;
		}
	}
	close;
}
