	callfunc "F_AuctionWarper",1;
}

	callfunc "F_AuctionWarper",3;

	callfunc "F_AuctionWarper",4;
}

function	script	F_AuctionWarper	{
	mes "[拍卖场向导]";
	mes "您好，请问您要进入拍卖场吗？";
	next;
	if (select("是:否") == 1) {
		.@num = getarg(0);
		if (.@num == 1 || .@num == 4) {
			mes "[拍卖场向导]";
			mes "太好了！那么，祝您玩得开心，好好享受拍卖的乐趣~";
		} else {
			mes "[拍卖场向导]";
			mes "祝您拍卖顺利。";
		}
		close2;
		switch(.@num) {
		case 1:
			warp "auction_01",179,53; end;
		case 2:
			warp "auction_01",21,43; end;
		case 3:
			warp "auction_02",151,23; end;
		case 4:
			warp "auction_02",43,24; end;
		}
	}
	mes "[拍卖场向导]";
	mes "好的，那么回头见。如果您改变主意，随时欢迎您";
	mes "来享受拍卖的乐趣~";
	close;
}
