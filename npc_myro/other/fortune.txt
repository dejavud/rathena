L_displaycutin:
	.@arg = getarg(0);
	if (.@arg < 4) cutin "绊胶飘傅墨靛",4;		// Ghostring Card
	else if (.@arg < 7) cutin "靛酚橇胶墨靛",4;	// Drops Card
	else if (.@arg < 10) cutin "盖萍胶墨靛",4;	// Mantis Card
	else if (.@arg < 13) cutin "公倡墨靛",4;		// Munak Card
	else if (.@arg < 16) cutin "魂鸥器傅墨靛",4;	// Santa Poring Card
	else if (.@arg < 19) cutin "贾廉胶奶沛墨靛",4;// Soldier Skeleton Card
	else if (.@arg < 22) cutin "胶奶饭沛墨靛",4;	// Skeleton Card
	else if (.@arg < 25) cutin "俊靛啊墨靛",4;	// Eddga Card
	else if (.@arg < 28) cutin "浚俩傅墨靛",4;	// Angeling Card
	else if (.@arg < 31) cutin "器傅墨靛",4;		// Poring Card
	else cutin "器器傅墨靛",4;// Poporing Card
	return;
}
