p_track01,58,0,0	script	竞速进度计时器	-1,{
OnEnable:
	initnpctimer;
	enablenpc "竞速进度计时器";
	for ( .@c = 1; .@c < 7; .@c++ ) {
		.@line = rand(1,70);
		enablenpc "starting#"+.@c;
		enablenpc "Tire"+.@c+"#1";
		if ( .@line <= 10 ) {
			.@tired = rand(50,60);
			enablenpc "Luk"+.@c+"#5";
			enablenpc "Luk"+.@c+"#6";
		} else if ( .@line <= 30 ) {
			.@tired = rand(40,60);
			enablenpc "Luk"+.@c+"#5";
			if (.@tired >= 50) enablenpc "Tire"+.@c+"#2";
		} else if ( .@line <= 40 ) {
			.@tired = rand(30,50);
			enablenpc "Luk"+.@c+"#1";
			enablenpc "Tire"+.@c+"#2";
			if ( .@tired < 40 ) enablenpc "Tire"+.@c+"#3";
		} else if (.@line <= 50) {
			.@tired = rand(20,40);
			enablenpc "Luk"+.@c+"#1";
			enablenpc "Luk"+.@c+"#2";
			enablenpc "Tire"+.@c+"#2";
			enablenpc "Tire"+.@c+"#3";
			if (.@tired < 30) enablenpc "Tire"+.@c+"#4";
		} else if (.@line <= 60) {
			.@tired = rand(10,30);
			for ( .@i = 1; .@i <= 3; .@i++ )
				enablenpc "Luk"+.@c+"#"+.@i;
			for ( .@i = 2; .@i <= 4; .@i++ )
				enablenpc "Tire"+.@c+"#"+.@i;
			if ( .@tired < 20) enablenpc "Tire"+.@c+"#5";
		} else if (.@line <= 70) {
			.@tired = rand(0,20);
			for ( .@i = 1; .@i <= 4; .@i++ )
				enablenpc "Luk"+.@c+"#"+.@i;
			for ( .@i = 2; .@i <= 5; .@i++ )
				enablenpc "Tire"+.@c+"#"+.@i;
			if (.@tired < 10) enablenpc "Tire"+.@c+"#6";
		}
		setd "$@mr_1_luk"+.@c, .@line;
		setd "$@mr_1_tire"+.@c, .@tired;
	}
	end;
OnTimer1000:
	enablenpc "门票助手#single";
	end;
OnTimer7000:
	mapannounce "p_track01","欢迎来到怪物竞速竞技场。",bc_map,"0xffb6c1";
	end;
OnTimer10000:
	mapannounce "p_track01","有任何问题，请随时向咨询台询问。",bc_map,"0xffb6c1";
	end;
OnTimer120000:
	mapannounce "p_track01","单只怪物竞速将在 3 分钟后开始。",bc_map,"0xffb6c1";
	end;
OnTimer123000:
	mapannounce "p_track01","如果您想为竞速下注，请咨询门票助手。",bc_map,"0xffb6c1";
	end;
OnTimer240000:
	mapannounce "p_track01","单只怪物竞速即将开始。",bc_map,"0xffb6c1";
	end;
OnTimer243000:
	mapannounce "p_track01","如果您想为竞速下注，请咨询门票助手。",bc_map,"0xffb6c1";
	end;
OnTimer300000:
	mapannounce "p_track01","怪物竞速已经开始。祝所有参加者好运。",bc_map,"0xffb6c1";
	$@monster_race = 0;
	disablenpc "门票助手#single";
