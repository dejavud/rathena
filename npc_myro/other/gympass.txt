payon,173,141,4	script	肌肉男 卡布斯#GymPass	899,{
	mes "[卡布斯]";
	mes "嘿，你好。现在的人可没有以前那么爱运动了。";
	mes "就算你是靠打架吃饭的，身体某些部位也可能又虚又软。";
	mes "懂我意思吧？";
	next;
	mes "[卡布斯]";
	mes "嘿，跟我一起练，我保证你能背得动、扛得起更多的东西。";
	mes "每次只要给我一张^FF0000健身卡^000000，咱们就可以开始了。";
	next;
	mes "[卡布斯]";
	mes "不过别高兴得太早：不管我带你练多少，都不能过度。";
	mes "你听说过有谁练得太壮吗？那是因为他们都累死了。懂？";
	next;
	mes "[卡布斯]";
	mes "照我说，你认真跟我训练，把物品携带能力提升^FF000010 次^000000";
	mes "是绝对安全的。那么……准备好挥汗如雨了吗？";
	next;
	switch(select("是:否:嗯，我的训练效果消失了。")) {
	case 1:
		if (gympassmemory < 10) {
			set .@add_carry,gympassmemory + 1;
			set .@remain_carry,10 - .@add_carry;
			if (countitem(7776) > 0) {
				mes "[卡布斯]";
				mes "噢，太棒了，我看到你带上健身卡了。好，照我的动作做，";
				mes "感受肌肉的燃烧。准备好了吗？开始。";
				next;
				specialeffect2 EF_EARTHSPIKE;
				next;
				specialeffect2 EF_DEVIL;
				next;
				specialeffect2 EF_COIN;
				next;
				specialeffect2 EF_SIGHTRASHER;
				next;
				mes "[卡布斯]";
				mes "好了，你现在应该能携带更多东西了。";
				mes "让我看看，像这样继续练下去，你的物品携带能力";
				mes "还能再提升^FF00000" + .@remain_carry + "^000000 次。";
				delitem 7776,1; //Max_Weight_Up_Scroll
				set gympassmemory,.@add_carry;
				skill "ALL_INCCARRY",.@add_carry,SKILL_PERM_GRANT;
				close;
			}
			else {
				mes "[卡布斯]";
				mes "老兄，我怎么说的？想锻炼、想练出能扛东西的肌肉，";
				mes "就得带上你的^FF0000健身卡^000000。";
				close;
			}
		}
		else {
			mes "[卡布斯]";
			mes "老兄，我看你的负重肌肉已经练到极限了。";
			mes "再练下去对你的身体太危险了！拜托，我跟你说过上限的。";
			close;
		}
	case 2:
		mes "[卡布斯]";
		mes "噢，那太可惜了。好吧，改变主意了再来找我。帮我";
		mes "跟朋友们宣传一下：谁要是虚胖，我就帮谁练出好身材。";
		close;
	case 3:
		if (gympassmemory > 0) {
			mes "[卡布斯]";
			mes "怎么回事？你竟然让你的负重肌肉就这么萎缩了？";
			mes "算你运气好，肌肉是有记忆的。";
			mes "重新练回来不需要那么长时间……";
			next;
			specialeffect2 EF_EARTHSPIKE;
			next;
			specialeffect2 EF_DEVIL;
			next;
			specialeffect2 EF_COIN;
			next;
			specialeffect2 EF_SIGHTRASHER;
			next;
			mes "[卡布斯]";
			mes "怎么样？你的负重肌肉这么快就回来了！";
			mes "这次可别再偷懒了，好吗？";
			skill "ALL_INCCARRY",gympassmemory,SKILL_PERM_GRANT;
			close;
		}
		else {
			mes "[卡布斯]";
			mes "呃……我们以前没有一起练过。这一点我很确定。";
			close;
		}
	}
}
