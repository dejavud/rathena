-	script	邮箱#dummy::MailBox	-1,{
	mes "[邮箱]";
	mes "使用邮箱服务需要支付 130 zeny。您要使用这项服务吗？";
	next;
	switch(select("是:否")) {
	case 1:
		mes "[邮箱]";
		if (Zeny < 130) {
			mes "抱歉，您的钱不够。使用邮箱服务需要支付 130 zeny。";
			close;
		}
		mes "谢谢，欢迎再来。";
		set Zeny, Zeny - 130;
		close2;
		openmail;
		end;
	case 2:
		mes "[邮箱]";
		mes "谢谢，欢迎再来。";
		close;
	}
}
