arug_dun01,1,2,1	script	Monster Controler1#aru	81,{

	set .@callwhere,rand(1,4);
	if (.@callwhere == 1) {
		monster "arug_dun01",150,340,"库布林",1980,1,"Monster Controler1#aru::OnMyMobDead";
	}
	else if (.@callwhere == 2) {
		monster "arug_dun01",232,293,"库布林",1980,1,"Monster Controler1#aru::OnMyMobDead";
	}
	else if (.@callwhere == 3) {
		monster "arug_dun01",156,167,"库布林",1980,1,"Monster Controler1#aru::OnMyMobDead";
	}
	else {
		monster "arug_dun01",243,56,"库布林",1980,1,"Monster Controler1#aru::OnMyMobDead";
	}
	end;

OnKill:
	killmonster "arug_dun01","Monster Controler1#aru::OnMyMObDead";
	end;

OnMyMobDead:
	if (mobcount("arug_dun01","Monster Controler1#aru::OnMyMObDead") == 0) {
		mapannounce "arug_dun01", "库布林：啊呃——！",bc_map,"0x99CC00";
		mapannounce "arug_dun01", "摩莱斯通：我好像听到库布林的惨叫了！！是谁在那里？库布林出什么事了？喂，说你呢！",bc_map,"0x99CC00";
		donpcevent "矮人#aru_gd::OnEnable";
	}
	end;
}
