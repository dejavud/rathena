function	script	F_GldDunSw	{
	set .@GID, GetCastleData(getarg(0),1);
	if (.@GID == 0) {
		mes "[ 回荡的声音 ]";
		mes " 『能够克服考验、展现出真正勇气的人……将会找到道路……』";
		close;
	}
	else {
		mes "[ 回荡的声音 ]";
		mes " 『只有能够展现真正勇气的人，才能接受这项考验。』";
		next;
		mes " ";
		mes "这里有一个小小的拉杆。你要拉动它吗？";
		next;

		if (select("拉动。:不拉。") == 1) {
			if (getcharid(2) == .@GID) {
				warp "gld_dun"+getarg(1),getarg(2),getarg(3);
				end;
			}
			mes " ";
			mes " 什么也没发生。";
		}
		return;
	}
}

// Castle 1 ===============================================
nguild_alde,212,181,0	script	开关#DunN01	111,{
	callfunc "F_GldDunSw","nguild_alde","02",32,122;
	close;
}
