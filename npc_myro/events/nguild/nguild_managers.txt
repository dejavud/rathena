function	script	F_GldManager	{
	.@npc_name$ = getarg(0);
	.@map_name$ = getarg(1);
	.@gid = getcastledata(.@map_name$,1);
	.@gm_name$ = getguildmaster(.@gid);

	mes "[ "+ .@npc_name$ +" ]";
	if (.@gid == 0){
		mes "我在等待我的主人。勇敢的冒险者，去追寻你的命运吧！";
		return;
	}
	if (getcharid(2) != .@gid){
		mes "我只听从 ^5533FF" + getguildmaster(.@gid) + "^000000 的命令！喂！你甚至都不是公会成员！！";
		mes "守护者在哪里？消灭这些入侵者！";
		return;
	}
	if (is_guild_leader() == false){
		mes "你不是 ^5533FF" + getguildmaster(.@gid) + "^000000！我只听从 ^5533FF" + getguildmaster(.@gid) + "^000000 的命令。";
		return 0;
	}

	mes "欢迎您，会长 ^5533FF" + getguildmaster(.@gid) + "^000000！我会尽全力协助您！";
	next;

	menu "雇佣 / 解雇卡普拉服务人员",M_Kaf, "进入宝物仓库",M_Treas, "取消",M_End;

	//===========================
	M_Kaf:
	//======
		mes "[ "+ .@npc_name$ +" ]";
		if (GetCastleData(.@map_name$,9) == 1) goto L_Dismiss;
		if (getgdskilllv(.@gid,10001) == 0){
			mes "会长，您还没有与卡普拉公司签订合约。";
			mes "要雇佣卡普拉，必须先学会公会技能 ^5533FF卡普拉契约^000000。";
			return;
		}

		L_Hire:
			mes "您要雇佣卡普拉服务吗？需要支付 ^5533FF10,000 zeny^000000……";
			next;
			menu "雇佣卡普拉。",-,"取消",sM_KafEnd;

				mes "[ "+ .@npc_name$ +" ]";
				if (Zeny < 10000){
					mes "会长，您的钱不够雇佣卡普拉。雇佣已取消。";
					return;
				}
				set Zeny,Zeny-10000;
				enablenpc "卡普拉服务人员#"+getarg(4);
				SetCastleData .@map_name$,9,1;
				mes "您已与卡普拉公司签订了合约。";
				next;
				cutin "kafra_01",2;
				mes "[ 卡普拉服务人员 ]";
				mes "您好！我来为您提供周到的服务！我会尽全力为您效劳的。";
				next;
				cutin "kafra_01",255;
				mes "[ "+ .@npc_name$ +" ]";
				//mes "Your employment contract lasts ^5533FF1 month^000000.  After this term is over you will have to create a new contract.";
				mes "我相信卡普拉服务人员会对公会成员有所帮助。";
				return;

			sM_KafEnd:
				mes "[ "+ .@npc_name$ +" ]";
				mes "遵命，会长。不过我建议您尽快雇佣一位卡普拉！";
				return;

		L_Dismiss:
			mes "您要解雇现在的卡普拉吗？";
			next;
			menu "解雇",-,"取消",sM_KafEnd2;

				cutin "kafra_01",2;
				mes "[ 卡普拉服务人员 ]";
				mes "我做错什么了吗？如果有，请您原谅我好吗？";
				next;
				menu "解雇",-,"取消",ssM_KafEnd2;

					mes "[ 卡普拉服务人员 ]";
					mes "很遗憾不能再为贵公会服务了……";
					next;
					disablenpc "卡普拉服务人员#"+getarg(4);
					SetCastleData .@map_name$,9,0;
					cutin "kafra_01",255;
					mes "[ "+ .@npc_name$ +" ]";
					mes "卡普拉已被解雇。不过……我们真的应该尽快再雇一位卡普拉！";
					return;
				ssM_KafEnd2:
					mes "[ 卡普拉服务人员 ]";
					mes "谢谢会长，我会努力的！^^。";
					cutin "kafra_01",255;
					return;
			sM_KafEnd2:
				mes "[ "+ .@npc_name$ +" ]";
				mes "会长，我认为应该留下现在的卡普拉服务人员，";
				mes "她已经在我们身边尽心尽力了。";
				return;

	//=========================
	M_Treas:
	//========
		mes "[ "+ .@npc_name$ +" ]";
		mes "您要去我们的宝物仓库吗？只有会长您才有资格进入那个房间。";
		next;
		menu "进入宝物仓库。",-,"取消",sM_TresEnd;

			mes "[ "+ .@npc_name$ +" ]";
			mes "请跟我走这条秘密通道。";
			mes "出来时必须拉动秘密开关才能出来。";
			next;
			warp .@map_name$,getarg(2),getarg(3);
			return;
		sM_TresEnd:
			mes "[ "+ .@npc_name$ +" ]";
			mes "宝物每天都会产出。";
			mes "一有机会就应该去收取，因为如果错过时机，它们可能会消失。";
			return;


	//==========================
	M_End:
	//=======
		mes "[ "+ .@npc_name$ +" ]";
		mes "遵命，会长。";
		return;
}
