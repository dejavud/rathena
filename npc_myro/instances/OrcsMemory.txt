gef_fild10,242,202,0	script	次元峡谷碎片	406,{

	set .@party_id,getcharid(1);
	set .@md_name$,"Orc's Memory";

	if (!instance_check_party(.@party_id,2,30,80)) {
		mes "只有等级在 ^ff000030 ~ 80^000000 之间的玩家才能进入这个副本。";
		close;
	}

	set .@orctime,checkquest(12059,PLAYTIME);
	if (.@orctime == -1) {
		if (is_party_leader() == true) {
			mes "队伍状态已确认。要预约进入兽人的记忆吗？";
			next;
			switch(select("预约进入兽人的记忆:进入副本:取消")) {
			case 1:
				if (instance_create(.@md_name$) < 0) {
					mes "队伍名称："+ getpartyname(.@party_id);
					mes "队长："+strcharinfo(0);
					mes "^0000ff兽人的记忆^000000 - 预约失败。";
					close;
				}
				mes "^0000ff兽人的记忆^000000 - 正在尝试预约入场";
				mes "预约成功后，如果想进入兽人的记忆，";
				mes "必须从菜单中选择「进入副本」。";
				close;
			case 2:
				callsub L_Enter,0;
			case 3:
				close;
			}
		}
		if(select("进入副本 兽人的记忆:取消") == 2)
			end;
		callsub L_Enter,1;
	} else if (.@orctime == 0 || .@orctime == 1) {
		mes "如果副本已经生成，就可以进入了。";
		next;
		if(select("进入副本 兽人的记忆:取消") == 2)
			close;
		callsub L_Enter,0;
	} else if (.@orctime == 2) {
		mes "^0000ff所有与兽人的记忆副本相关的记录和后遗症都已清除。";
		mes "现在可以重新生成或再次进入副本了。^000000";
		erasequest 12059;
		close;
	}
	mes "要生成副本，你必须是队长，并且队伍中至少有 2 名成员。";
	close;
L_Enter:
	switch(instance_enter("Orc's Memory")) {
	case IE_OTHER:
		mes "发生了未知错误。";
		close;
	case IE_NOINSTANCE:
		mes "纪念副本「兽人的记忆」不存在。";
		mes "纪念副本已被队长解散，或者因为超时而消失了。";
		mes "请在 2 小时后再试。";
		close;
	case IE_NOMEMBER:
		mes "只有队伍成员才能进入纪念副本。";
		close;
	case IE_OK:
		mapannounce "gef_fild10", getpartyname( getcharid(1) ) +" 队伍的成员 "+strcharinfo(0)+" 进入了兽人的记忆。",bc_map,"0x00ff99";
		if (checkquest(12059) == -1) setquest 12059;
		//warp "1@orcs",179,15;
		if (getarg(0) == 0) close;
		else end;
	}

}

1@orcs,180,30,0	script	#Resurrect Monsters1	844,{
	end;

	OnInstanceInit:
	disablenpc instance_npcname("#Resurrect Monsters1");
	end;

	OnDisable:
	stopnpctimer;
	killmonster instance_mapname("1@orcs"),instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
	end;

	OnEnable:
	enablenpc instance_npcname("#Resurrect Monsters1");
	monster instance_mapname("1@orcs"),0,0,"兽人战士",1023,30,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
	end;

	OnMyMobDead:
	set .@map$, instance_mapname("1@orcs");
	set .@mob_dead_num, 30 - mobcount(.@map$,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead");
	if (.@mob_dead_num > 0) {
		set .@mob_ran,rand(1,30);
		if (.@mob_ran > 29) {
			monster .@map$,0,0,"兽人战士",1023,.@mob_dead_num,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
		}
		else if ((.@mob_ran > 28) && (.@mob_ran < 30)) {
			monster .@map$,0,0,"高级兽人",1213,.@mob_dead_num,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
			if (rand(1,10) == 9) {
				mapannounce .@map$, "高级兽人：我们需要更多的防御！再多调些人过来！",bc_map,"0xff4444";
			}
		}
		else if ((.@mob_ran > 26) && (.@mob_ran < 29)) {
			areamonster .@map$,41,91,51,81,"高级兽人",1213,.@mob_dead_num,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
			if (rand(1,10) == 9) {
				mapannounce .@map$, "高级兽人在哪里！？快让它们去挡住敌人！",bc_map,"0xff4444";
			}
		}
		else {
			areamonster .@map$,17,187,27,177,"高级兽人",1213,.@mob_dead_num,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
			if (rand(1,5) == 3) {
				mapannounce .@map$, "注意：部队开始在 4 号区域集结。",bc_map,"0x77ff77";
			}
			if (rand(1,100) == 50) {
				initnpctimer;
			}
		}
	}
	end;

	OnTimer10:
	mapannounce instance_mapname("1@orcs"), "警备队长：看来这会比预想的更费时间。召唤钟乳石魔像！",bc_map,"0xff4444";
	end;

	OnTimer4010:
	set .@map$, instance_mapname("1@orcs");
	mapannounce .@map$, "钟乳石魔像正从地底深处钻出来。",bc_map,"0x77ff77";
	areamonster .@map$,17,187,27,177,"钟乳石魔像",1278,20,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
	stopnpctimer;
	end;
}
