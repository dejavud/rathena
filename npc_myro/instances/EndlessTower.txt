e_tower,81,105,0	script	守护塔之石	406,{

	set .@party_id,getcharid(1);
	set .@md_name$,"Endless Tower";

	set .@etower_timer,checkquest(60200,PLAYTIME); // 1 week
	set .@etower_timer2,checkquest(60201,PLAYTIME); // 4 hours

	if (!instance_check_party(.@party_id,2)) {
		mes "请创建或加入一个 2 人以上的队伍，然后再试一次。";
		close;
	}
	switch(.@etower_timer) {
	case -1:
		if (is_party_leader() == true) {
			mes "已确认队伍创建完毕。要预约进入无尽之塔吗？";
			next;
			switch(select("生成副本 无尽之塔:进入副本:返回艾尔贝塔:取消")) {
			case 1:
				if (instance_create(.@md_name$) < 0) {
					mes "队伍名称："+ getpartyname(.@party_id);
					mes "队长："+strcharinfo(0);
					mes "^0000ff无尽之塔^000000 - 预约失败！";
					close;
				}
				mes "^0000ff无尽之塔^000000 - 尝试预约";
				mes "预约成功后，";
				mes "必须与后面的 NPC 对话并选择「进入副本」菜单才能进入副本。";
				close;
			case 2:
				callsub L_Enter,0,1;
			case 3:
				mes "我将把你传送到艾尔贝塔。";
				close2;
				warp "alberta",223,36;
				end;
			case 4:
				close;
			}
		}
		switch(select("进入无尽之塔:返回艾尔贝塔:取消")) {
		case 1:
			callsub L_Enter,1,1;
		case 2:
			mes "我将把你传送到艾尔贝塔。";
			close2;
			warp "alberta",223,36;
			end;
		case 3:
			end;
		}
	case 0:
	case 1:
		if (.@etower_timer2 < 2 && getcharid(1) == etower_partyid) {
			mes "如果副本已经生成，就可以进入了。";
			next;
			switch(select("进入无尽之塔:返回艾尔贝塔:取消")) {
			case 1:
				callsub L_Enter,0,0;
			case 2:
				mes "我将把你传送到艾尔贝塔。";
				close2;
				warp "alberta",223,36;
				end;
			case 3:
				close;
			}
		} else {
			set .@dun_lim_time,etower_timer+604800; // 1 week
			// set .@dun_lim_time2,etower_timer+14400; // 4 hours
			set .@dun_cur_time,gettimetick(2);
			set .@dun_ent_t,(.@dun_lim_time - .@dun_cur_time);
			set .@dun_h,(.@dun_ent_t / 3600);
			set .@dun_m,(.@dun_ent_t - (.@dun_h * 3600)) / 60;
			set .@dun_s,.@dun_ent_t - ((.@dun_h * 3600) + (.@dun_m * 60));

			mes "由于塔的后遗症，你现在无法进入副本。";
			mes "距离下次进入副本还剩 " + .@dun_h + " 小时 " + .@dun_m + " 分 " + .@dun_s + " 秒。";
			next;
			mes "这里很危险。让我把你传送到艾尔贝塔吧。";
			close2;
			warp "alberta",223,36;
			end;
		}
	case 2:
		set etower_timer,0;
		erasequest 60200;
		erasequest 60201;
		mes "^0000ff与无尽之塔相关的记录和后遗症都已清除。";
		mes "你可以重新生成并进入无尽之塔了。^000000";
		close;
	}

L_Enter:
	switch(instance_enter("Endless Tower")) {
	case IE_OTHER:
		mes "发生了未知错误。";
		close;
	case IE_NOINSTANCE:
		mes "纪念副本「无尽之塔」不存在。";
		mes "队长还没有生成副本。";
		close;
	case IE_NOMEMBER:
		mes "你必须先创建队伍才能进入副本。";
		close;
	case IE_OK:
		mapannounce "e_tower", strcharinfo(0) +"，队伍 "+ getpartyname( getcharid(1) ) +" 的成员，正在进入副本「无尽之塔」。",bc_map,"0x00ff99",FW_NORMAL,12;
		if (getarg(1)) {
			set etower_timer,gettimetick(2);
			set etower_partyid, getcharid(1);
			setquest 60200;
			setquest 60201;
		}
		//warp "1@tower",52,354;
		if (getarg(0) == 0) close;
		else end;
	}
}
